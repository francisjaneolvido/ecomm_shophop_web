<?php

namespace App\Http\Controllers;

use App\Models\Logistics\Delivery;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\Storage;
use Symfony\Component\HttpFoundation\Response;

class DeliveryProofController extends Controller
{
    public function show(Request $request, int $delivery): Response
    {
        // Proof is a private file and only parties to this Seller Order or assigned Rider may fetch it.
        $record = Delivery::with('order')->findOrFail($delivery);
        abort_unless($record->status === 'delivered' && $record->proof_path, 404);
        $user = Auth::guard('web')->user();
        $rider = Auth::guard('rider')->user()?->fresh(['partner.user']);
        $allowed = ! ($user && $rider) && (
            ($rider && $rider->status === 'active' && $rider->id === $record->rider_id
                && $rider->logistics_partner_id === $record->logistics_partner_id
                && $rider->partner?->user?->status === 'approved')
            || ($user && $user->status === 'approved' && match ($user->account_type) {
                'buyer' => $user->buyer?->id === $record->order?->buyer_id,
                'seller' => $user->seller?->id === $record->order?->seller_id,
                'logistics' => $user->logisticsPartner?->id === $record->logistics_partner_id,
                default => false,
            })
        );
        abort_unless($allowed, 403);
        abort_unless(Storage::disk('local')->exists($record->proof_path), 404);

        return Storage::disk('local')->response($record->proof_path);
    }
}

<?php

namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use App\Models\User;
use App\Models\Logistics\BranchStaff;
use App\Services\ParcelScanService;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\Validation\ValidationException;

/** Logistics registration stays on Web; approved company account may use the scanning-only mobile workspace. */
class MobileLogisticsScannerController extends Controller
{
    public function branches(Request $request): JsonResponse
    {
        $partner = $this->partner($request);
        $actor = $request->user();
        $centers = $partner->sortingCenters()->where('status', 'active');
        if ($actor instanceof BranchStaff) $centers->whereKey($actor->sorting_center_id);
        return response()->json(['success' => true, 'status' => 'ok', 'message' => 'Active sorting centers loaded.', 'data' => [
            'company' => $partner->company_name,
            'operator' => $actor instanceof BranchStaff ? $actor->name : $partner->company_name,
            'branch_locked' => $actor instanceof BranchStaff,
            'branches' => $centers->orderByDesc('is_main')->orderBy('name')->get(['id','code','name','municipality','province','region']),
        ]]);
    }

    /** Confirm eligibility before a staff member physically accepts a parcel. */
    public function preview(Request $request, ParcelScanService $scanner): JsonResponse
    {
        $partner = $this->partner($request);
        $data = $request->validate([
            'sorting_center_id' => ['required', 'integer'],
            'tracking_code' => ['required', 'string', 'max:60'],
        ]);
        $this->requireBranchAccess($request, $partner, (int) $data['sorting_center_id']);
        try {
            $parcel = $scanner->inspect((int) $partner->id, (int) $data['sorting_center_id'], $data['tracking_code']);
        } catch (ValidationException $exception) {
            return response()->json(['success' => false, 'status' => 'scan_conflict',
                'message' => collect($exception->errors())->flatten()->first() ?: 'Parcel cannot be received.'], 409);
        }
        return response()->json(['success' => true, 'status' => 'ready',
            'message' => 'Verify the physical parcel before confirming receipt.', 'data' => ['parcel' => $parcel]]);
    }

    /** Last scans at an authorized branch only; no buyer address or payment details. */
    public function history(Request $request): JsonResponse
    {
        $partner = $this->partner($request);
        $data = $request->validate(['sorting_center_id' => ['required', 'integer']]);
        $branchId = (int) $data['sorting_center_id'];
        $this->requireBranchAccess($request, $partner, $branchId);

        $entries = DB::table('parcel_scan_events as scans')
            ->join('deliveries as deliveries', 'deliveries.id', '=', 'scans.delivery_id')
            ->where('deliveries.logistics_partner_id', $partner->id)
            ->where('scans.sorting_center_id', $branchId)
            ->orderByDesc('scans.scanned_at')
            ->limit(30)
            ->get(['deliveries.tracking_code', 'scans.event_type', 'scans.scanned_at'])
            ->map(fn ($row) => [
                'tracking_code' => $row->tracking_code,
                'event' => $row->event_type,
                'scanned_at' => $row->scanned_at,
            ]);

        return response()->json(['success' => true, 'status' => 'ok',
            'message' => 'Recent branch scans loaded.', 'data' => ['events' => $entries]]);
    }

    private function requireBranchAccess(Request $request, $partner, int $branchId): void
    {
        abort_unless($partner->sortingCenters()->where('status', 'active')->whereKey($branchId)->exists(),
            403, 'This is not an active branch of your Logistics company.');
        $actor = $request->user();
        if ($actor instanceof BranchStaff) {
            abort_unless($actor->activeForBranch((int) $partner->id, $branchId),
                403, 'Your staff account cannot access this branch.');
        }
    }

    public function receive(Request $request, ParcelScanService $scanner): JsonResponse
    {
        $partner = $this->partner($request);
        $actor = $request->user();
        $data = $request->validate([
            'sorting_center_id' => ['required','integer'],
            'tracking_code' => ['required','string','max:60'],
        ]);

        $this->requireBranchAccess($request, $partner, (int) $data['sorting_center_id']);
        try {
            $result = $scanner->receive(
                (int) $partner->id,
                (int) $data['sorting_center_id'],
                $actor instanceof User ? (int) $actor->id : (int) $partner->user_id,
                $data['tracking_code'],
                $actor instanceof BranchStaff ? $actor->id : null,
            );
        } catch (ValidationException $exception) {
            return response()->json([
                'success' => false, 'status' => 'scan_conflict',
                'message' => collect($exception->errors())->flatten()->first() ?: 'Cannot receive parcel.',
                'errors' => $exception->errors(),
            ], 409);
        }

        return response()->json([
            'success' => true, 'status' => 'received',
            'message' => $result['event'] === 'pickup_received'
                ? 'Pickup parcel received and logged at this branch.'
                : 'Inter-branch transfer received and logged.',
            'data' => $result,
        ]);
    }

    private function partner(Request $request)
    {
        $user = $request->user();
        if ($user instanceof BranchStaff) {
            abort_unless($user->status === 'active'
                && $user->sortingCenter?->status === 'active'
                && $user->partner?->user?->status === 'approved', 403, 'Inactive staff or branch.');
            abort_unless((int) $user->sortingCenter->logistics_partner_id === (int) $user->logistics_partner_id,
                403, 'Invalid staff branch.');
            return $user->partner;
        }
        abort_unless($user instanceof User && $user->account_type === 'logistics'
            && $user->status === 'approved', 403, 'Approved Logistics company account required.');
        $partner = $user->logisticsPartner;
        abort_unless($partner, 403, 'Logistics profile unavailable.');
        return $partner;
    }
}

<?php

namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use App\Models\Buyer\Order\Order;
use App\Models\Finance\SettlementEntry;
use App\Models\Logistics\Delivery;
use App\Models\Logistics\ParcelTransfer;
use App\Models\Logistics\Rider;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;

class MobileRiderTransitController extends Controller
{
    public function index(Request $request): JsonResponse
    {
        $rider = $this->rider($request);
        $transfers = ParcelTransfer::with(['delivery.order', 'fromCenter', 'toCenter'])
            ->where('transit_rider_id', $rider->id)
            ->whereHas('delivery', fn ($query) => $query->where('logistics_partner_id', $rider->logistics_partner_id))
            ->latest()->limit(100)->get();
        return response()->json(['success' => true, 'status' => 'ok', 'message' => 'Transit assignments loaded.', 'data' => [
            'transfers' => $transfers->map(fn ($transfer) => [
                'id' => $transfer->id,
                'tracking_code' => $transfer->delivery?->tracking_code,
                'status' => $transfer->status,
                'departed_at' => optional($transfer->departed_at)?->toIso8601String(),
                'received_at' => optional($transfer->received_at)?->toIso8601String(),
                'from' => $transfer->fromCenter?->name,
                'to' => $transfer->toCenter?->name,
                'can_depart' => $transfer->status === ParcelTransfer::IN_TRANSIT && ! $transfer->departed_at,
            ])->values(),
        ]]);
    }

    public function depart(Request $request, int $transfer): JsonResponse
    {
        $rider = $this->rider($request);
        return DB::transaction(function () use ($rider, $transfer) {
            $job = ParcelTransfer::with('delivery.order')->where('transit_rider_id', $rider->id)
                ->whereKey($transfer)->lockForUpdate()->firstOrFail();
            $delivery = Delivery::where('logistics_partner_id', $rider->logistics_partner_id)
                ->whereKey($job->delivery_id)->lockForUpdate()->firstOrFail();
            if ($job->status !== ParcelTransfer::IN_TRANSIT || $job->departed_at
                || (int) $job->from_sorting_center_id !== (int) $rider->sorting_center_id
                || (int) $job->from_sorting_center_id !== (int) $delivery->current_sorting_center_id
                || $delivery->status !== Delivery::SORTED
                || $delivery->order?->status !== Order::STATUS_TO_RECEIVE) {
                return response()->json(['success' => false, 'status' => 'stale',
                    'message' => 'Transit assignment changed or is not ready. Refresh your tasks.'], 409);
            }
            $job->update(['departed_at' => now()]);
            return response()->json(['success' => true, 'status' => 'departed',
                'message' => 'Transit departure confirmed. Destination branch must scan the parcel on arrival.',
                'data' => ['transfer_id' => $job->id]]);
        });
    }

    public function earnings(Request $request): JsonResponse
    {
        $rider = $this->rider($request);
        $entries = SettlementEntry::where('payee_id', $rider->id)
            ->where(function ($query) { $query->whereIn('kind', ['pickup_rider', 'delivery_rider'])->orWhere('kind', 'like', 'transit_%'); })
            ->whereHas('order.delivery', fn ($query) => $query->where('logistics_partner_id', $rider->logistics_partner_id))
            ->latest()->limit(100)->get();
        return response()->json(['success' => true, 'status' => 'ok', 'message' => 'Rider earnings loaded.', 'data' => [
            'pending' => round((float) $entries->where('status', 'pending')->sum('amount'), 2),
            'paid' => round((float) $entries->where('status', 'paid')->sum('amount'), 2),
            'entries' => $entries->map(fn ($entry) => [
                'order_id' => $entry->order_id, 'kind' => $entry->kind, 'amount' => (float) $entry->amount,
                'status' => $entry->status, 'paid_at' => optional($entry->paid_at)?->toIso8601String(),
            ])->values(),
        ]]);
    }

    private function rider(Request $request): Rider
    {
        $actor = $request->user();
        abort_unless($actor instanceof Rider, 403, 'Rider login required.');
        $rider = $actor->fresh(['partner.user']);
        abort_unless($rider && $rider->status === 'active' && $rider->email_verified_at
            && $rider->partner?->user?->status === 'approved', 403, 'Active Rider required.');
        return $rider;
    }
}

<?php

// app/Http/Controllers/branch/ParcelController.php

namespace App\Http\Controllers\branch;

use App\Http\Controllers\Controller;
use App\Models\Logistics\Delivery;
use App\Models\Logistics\Rider;
use App\Models\Logistics\SortingCenter;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\View\View;

class ParcelController extends Controller
{
    /**
     * Tab key sa view => status values sa deliveries table.
     * ASSUMPTION: i-adjust sa totoong status values ng Delivery model mo.
     */
    private const STATUS_MAP = [
        'at_branch'        => ['at_branch', 'sorted'],
        'out_for_delivery' => ['out_for_delivery'],
        'delivered'        => ['delivered'],
        'failed'           => ['failed', 'failed_attempt'],
        'returning'        => ['returning', 'return_to_sender'],
    ];

    /** Mga status na pwedeng i-assign sa rider. */
    private const ASSIGNABLE = ['at_branch', 'sorted', 'failed', 'failed_attempt'];

    public function index(Request $request): View
    {
        $center = $this->currentCenter($request);

        $status = $request->query('status', 'all');
        if ($status !== 'all' && ! array_key_exists($status, self::STATUS_MAP)) {
            $status = 'all';
        }
        $search = trim((string) $request->query('q', ''));

        $base = Delivery::query()->where('sorting_center_id', $center->id);

        // Stats (isang query, hindi apektado ng filter/search)
        $rows = (clone $base)->select('status', DB::raw('COUNT(*) as total'))
            ->groupBy('status')->pluck('total', 'status');

        $countFor = fn (string $key) => collect(self::STATUS_MAP[$key])
            ->sum(fn ($s) => (int) ($rows[$s] ?? 0));

        $counts = [
            'at_branch'        => $countFor('at_branch'),
            'out_for_delivery' => $countFor('out_for_delivery'),
            'failed'           => $countFor('failed'),
        ];

        $parcels = $base
            ->with(['order', 'rider'])
            ->when($status !== 'all', fn ($q) => $q->whereIn('status', self::STATUS_MAP[$status]))
            ->when($search !== '', function ($q) use ($search) {
                $like = '%'.str_replace(['%', '_'], ['\%', '\_'], $search).'%';
                $q->where(function ($w) use ($like) {
                    $w->where('tracking_number', 'like', $like)
                      ->orWhere('receiver_name', 'like', $like)
                      ->orWhereHas('order', fn ($o) => $o->where('order_number', 'like', $like));
                });
            })
            ->latest('updated_at')
            ->paginate(20)
            ->withQueryString()
            ->through(fn (Delivery $d) => $this->present($d));

        $riders = Rider::query()
            ->where('sorting_center_id', $center->id)
            ->where('status', 'active')
            ->orderBy('name')
            ->get(['id', 'name']);

        return view('branch.parcels.index', [
            'center'       => $center,
            'parcels'      => $parcels,
            'counts'       => $counts,
            'riders'       => $riders,
            'activeStatus' => $status,
            'search'       => $search,
        ]);
    }

    public function assign(Request $request): RedirectResponse
    {
        $center = $this->currentCenter($request);

        $data = $request->validate([
            'parcel_ids'   => ['required', 'array', 'min:1', 'max:100'],
            'parcel_ids.*' => ['string', 'max:64'],
            'rider_id'     => ['required', 'integer'],
        ], [
            'parcel_ids.required' => 'Pumili muna ng kahit isang parcel.',
            'rider_id.required'   => 'Pumili ng rider.',
        ]);

        // Dapat rider ng branch na ito at active
        $rider = Rider::query()
            ->where('sorting_center_id', $center->id)
            ->where('status', 'active')
            ->find($data['rider_id']);

        if (! $rider) {
            return back()->with('error', 'Hindi valid ang rider na napili.');
        }

        $assigned = DB::transaction(function () use ($center, $rider, $data) {
            $deliveries = Delivery::query()
                ->where('sorting_center_id', $center->id)
                ->whereIn('tracking_number', $data['parcel_ids'])
                ->whereIn('status', self::ASSIGNABLE)
                ->lockForUpdate()
                ->get();

            foreach ($deliveries as $delivery) {
                $delivery->update([
                    'rider_id' => $rider->id,
                    'status'   => 'out_for_delivery',
                ]);
            }

            return $deliveries->count();
        });

        if ($assigned === 0) {
            return back()->with('error', 'Walang na-assign. Baka na-update na ang mga parcel.');
        }

        $skipped = count($data['parcel_ids']) - $assigned;
        $message = "{$assigned} parcel(s) na-assign kay {$rider->name}.";
        if ($skipped > 0) {
            $message .= " {$skipped} ang nilaktawan dahil hindi na pwedeng i-assign.";
        }

        return back()->with('success', $message);
    }

    /** Hugis ng data na hinihingi ng branch.parcels.index view. */
    private function present(Delivery $d): array
    {
        $viewStatus = collect(self::STATUS_MAP)
            ->search(fn ($values) => in_array($d->status, $values, true)) ?: 'at_branch';

        return [
            'tracking' => $d->tracking_number,
            'order'    => $d->order->order_number ?? '—',
            'receiver' => $d->receiver_name,
            'barangay' => $d->delivery_address,
            'cod'      => (float) ($d->cod_amount ?? 0),
            'status'   => $viewStatus,
            'rider'    => $d->rider->name ?? null,
            'updated'  => optional($d->updated_at)->format('M d, g:i A'),
            'note'     => $d->failure_note ?? null,
        ];
    }

    /**
     * ASSUMPTION: paano malaman ang branch ng naka-login.
     * Palitan ayon sa auth setup mo (guard, relation, o session).
     */
    private function currentCenter(Request $request): SortingCenter
    {
        $user = $request->user();

        $center = SortingCenter::query()
            ->where('id', $user->sorting_center_id ?? null)
            ->first();

        abort_unless($center, 403, 'Walang naka-link na branch sa account mo.');

        return $center;
    }
}

    public function placeOrder(Request $request)
    {
        $buyer = Auth::user()->buyer;

        $data = $request->validate([
            'items' => 'required|array|min:1',
            'items.*.quantity' => 'required|integer|min:1',
            'shipping_method' => 'required|array',

    public function reconcile(Request $request, int $delivery): RedirectResponse
    {
        $partner = $this->partner($request);

        return DB::transaction(function () use ($request, $partner, $delivery) {
            // Partner ownership is derived from the authenticated User, never a submitted partner or amount.
            abort_unless(User::whereKey($request->user()->id)->where('account_type', 'logistics')
                ->where('status', 'approved')->lockForUpdate()->first(), 403);
            Delivery::whereKey($delivery)->where('logistics_partner_id', $partner->id)->firstOrFail();
            $cash = CodSettlement::where('delivery_id', $delivery)
                ->where('logistics_partner_id', $partner->id)->lockForUpdate()->first();
            if (! $cash || $cash->status !== CodSettlement::REMITTED || ! $cash->remitted_at
                || $cash->reconciled_at || $cash->expected_amount !== $cash->collected_amount) {
                return back()->withErrors(['settlement' => 'Cash is not ready for receipt confirmation. Refresh and review it.']);
            }
            if (CodSettlement::whereKey($cash->id)->where('status', CodSettlement::REMITTED)
                ->whereNull('reconciled_at')->update([
                    'status' => CodSettlement::RECONCILED,
                    'received_amount' => $cash->collected_amount,
                    'reconciled_by_user_id' => $request->user()->id,
                    'reconciled_at' => now(),
                ]) !== 1) {
                return back()->withErrors(['settlement' => 'Cash receipt changed. Refresh and review it.']);
            }

            return redirect()->route('logistics.settlements.index')->with('status', 'Recorded COD cash receipt confirmed.');
        });
    }

    /** Logistics records manual ShopHop remittance; Admin must independently confirm receipt. */
    public function remitToAdmin(Request $request, int $delivery): RedirectResponse
    {
        $data = $request->validate(['reference' => ['required','string','min:4','max:120']]);
        $partner = $this->partner($request);
        DB::transaction(function () use ($partner, $delivery, $data) {
            $cash = CodSettlement::where('delivery_id', $delivery)
                ->where('logistics_partner_id', $partner->id)->lockForUpdate()->firstOrFail();
            if ($cash->status !== CodSettlement::RECONCILED || ! $cash->reconciled_at
                || $cash->admin_remitted_at || $cash->admin_received_at
                || abs((float) $cash->received_amount - (float) $cash->expected_amount) > .001) {
                throw \Illuminate\Validation\ValidationException::withMessages([
                    'settlement' => 'COD must be reconciled and not already remitted to ShopHop.']);
            }
            $cash->update(['admin_remittance_reference' => trim($data['reference']),
                'admin_remitted_at' => now()]);
        });
        return back()->with('status', 'Remittance to ShopHop recorded; pending Admin receipt confirmation.');
    }

    /** Assign the rider who will collect the parcel from the Seller and bring it to the sorting center. */
    public function assign(Request $request, int $order): RedirectResponse
    {
        $partner = $this->partner($request);
        $data = $request->validate(['rider_id' => ['required', 'integer']]);
        $rider = $this->activeRider($partner, (int) $data['rider_id']);

        try {
            return DB::transaction(function () use ($partner, $rider, $order) {
                $ready = Order::with('pickupRequest')->whereKey($order)->lockForUpdate()->firstOrFail();
                $pickup = PickupRequest::where('order_id', $ready->id)->lockForUpdate()->first();
                if ($ready->status !== Order::STATUS_READY_FOR_PICKUP || ! $ready->isPaymentEligible()
                    || ! $pickup || (int) $pickup->logistics_partner_id !== (int) $partner->id
                    || $pickup->status !== PickupRequest::REQUESTED
                    || Delivery::where('order_id', $order)->exists()
                    || ! $this->riderStillActive($rider, $partner)) {
                    return $this->stale();
                }

                $originCenterId = $pickup->origin_sorting_center_id
                    ?: $rider->sorting_center_id
                    ?: $partner->mainSortingCenter()->value('id')
                    ?: $partner->sortingCenters()->where('status', 'active')->value('id');

                if ($originCenterId && $rider->sorting_center_id && (int) $rider->sorting_center_id !== (int) $originCenterId) {
                    throw ValidationException::withMessages([
                        'rider_id' => 'Choose a pickup Rider assigned to the requested origin Sorting Center.',
                    ]);
                }

                Delivery::create([
                    'order_id' => $ready->id,
                    'tracking_code' => $ready->tracking_code ?: $this->trackingCode($ready),
                    'logistics_partner_id' => $partner->id,
                    'origin_sorting_center_id' => $originCenterId,
                    'rider_id' => $rider->id,
                    'status' => Delivery::PICKUP_ASSIGNED,
                    'assigned_at' => now(),
                ]);

                $pickup->update([
                    'status' => PickupRequest::ASSIGNED,
                    'origin_sorting_center_id' => $originCenterId,
                    'assigned_at' => now(),
                ]);

                return redirect()->route('logistics.deliveries.board')
                    ->with('status', 'Pickup Rider assigned.');
            });
        } catch (QueryException $exception) {
            if (Delivery::where('order_id', $order)->exists()) {

    /** Record the destination area selected after reading the delivery address. */
    public function sort(Request $request, int $delivery): RedirectResponse
    {

        $data = $request->validate(['destination_area_id' => ['required', 'integer']]);
        $area = LogisticsCoverageArea::where('logistics_partner_id', $partner->id)

                ?: $partner->mainSortingCenter()->value('id');

            $owned->update([
                'destination_area_id' => $area->id,
                'destination_sorting_center_id' => $destinationCenterId,
                'status' => Delivery::SORTED,
                'sorted_at' => now(),

    /** Assign the rider responsible for final delivery after sorting, or after a failed attempt. */
    public function assignDelivery(Request $request, int $delivery): RedirectResponse
    {
        $partner = $this->partner($request);
        $data = $request->validate(['rider_id' => ['required', 'integer']]);
        $rider = $this->activeRider($partner, (int) $data['rider_id']);

        return DB::transaction(function () use ($partner, $delivery, $rider) {
            $owned = Delivery::with(['order', 'destinationArea', 'transfers'])
                ->where('logistics_partner_id', $partner->id)
                ->whereKey($delivery)->lockForUpdate()->firstOrFail();

            if (! in_array($owned->status, [Delivery::SORTED, Delivery::DELIVERY_FAILED], true)
                || $owned->order?->status !== Order::STATUS_TO_RECEIVE
                || $owned->returnCase()->exists()
                || ! $owned->destination_area_id || ! $this->riderStillActive($rider, $partner)) {
                return $this->stale();
            }

            if ((int) $rider->coverage_area_id !== (int) $owned->destination_area_id) {
                throw ValidationException::withMessages([
                    'rider_id' => 'Choose an active Rider assigned to this parcel\'s destination area.',
                ]);
            }

            if ($owned->destination_sorting_center_id
                && (int) $owned->current_sorting_center_id !== (int) $owned->destination_sorting_center_id) {
                throw ValidationException::withMessages([
                    'rider_id' => 'Receive the parcel at its destination Sorting Center before assigning the final-delivery Rider.',
                ]);
            }

            if ($owned->destination_sorting_center_id
                && (int) $rider->sorting_center_id !== (int) $owned->destination_sorting_center_id) {
                throw ValidationException::withMessages([
                    'rider_id' => 'Choose a Rider assigned to the destination Sorting Center.',
                ]);
            }

            if ($owned->transfers->contains(fn ($transfer) => $transfer->status === ParcelTransfer::IN_TRANSIT)) {
                throw ValidationException::withMessages([
                    'rider_id' => 'This parcel is still in transit between Sorting Centers.',
                ]);
            }

            $owned->update([
                'rider_id' => $rider->id,
                'delivery_rider_id' => $rider->id,
                'status' => Delivery::DELIVERY_ASSIGNED,
                'delivery_assigned_at' => now(),
                'delivery_attempts' => $owned->delivery_attempts + 1,
            ]);

            return back()->with('status', 'Delivery Rider assigned for the destination area.');
        });
    }

    /** Dispatch a sorted parcel from its current center to the branch that owns the destination area. */
    public function dispatchTransfer(Request $request, int $delivery): RedirectResponse
    {
        $partner = $this->partner($request);
        $data = $request->validate([
            'to_sorting_center_id' => ['required', 'integer'],
            'transit_rider_id' => ['nullable', 'integer'],
        ]);

        return DB::transaction(function () use ($partner, $delivery, $data) {
            $owned = Delivery::with(['order', 'transfers'])
                ->where('logistics_partner_id', $partner->id)
                ->whereKey($delivery)->lockForUpdate()->firstOrFail();

            if ($owned->status !== Delivery::SORTED
                || ! $owned->current_sorting_center_id
                || ! $owned->destination_sorting_center_id
                || (int) $owned->current_sorting_center_id === (int) $owned->destination_sorting_center_id) {
                return $this->stale();
            }

            if ($owned->transfers->contains(fn ($transfer) => $transfer->status === ParcelTransfer::IN_TRANSIT)) {
                return back()->withErrors(['delivery' => 'This parcel already has an active center transfer.']);
            }

            $from = SortingCenter::where('logistics_partner_id', $partner->id)->findOrFail($owned->current_sorting_center_id);
            $to = SortingCenter::where('logistics_partner_id', $partner->id)
                ->where('status', 'active')
                ->findOrFail((int) $data['to_sorting_center_id']);

            if ((int) $from->id === (int) $to->id) {
                throw ValidationException::withMessages([
                    'to_sorting_center_id' => 'Choose a different next Sorting Center.',
                ]);
            }

            $transitRider = null;
            if (! empty($data['transit_rider_id'])) {
                $transitRider = $this->activeRider($partner, (int) $data['transit_rider_id']);
                if ((int) $transitRider->sorting_center_id !== (int) $from->id) {
                    throw ValidationException::withMessages([
                        'transit_rider_id' => 'Transit Rider must belong to the dispatching branch.',
                    ]);
                }
            }

            ParcelTransfer::create([
                'delivery_id' => $owned->id,
                'from_sorting_center_id' => $from->id,

    public function receiveTransfer(Request $request, int $transfer, ParcelScanService $scanner): RedirectResponse
    {
        $partner = $this->partner($request);
        $data = $request->validate(['tracking_code' => ['required', 'string', 'max:60']]);
        $parcelTransfer = ParcelTransfer::with('delivery')->findOrFail($transfer);
        abort_unless((int) $parcelTransfer->delivery?->logistics_partner_id === (int) $partner->id, 404);
        abort_unless($parcelTransfer->status === ParcelTransfer::IN_TRANSIT, 422, 'Transfer is not in transit.');
        if (! hash_equals((string) $parcelTransfer->delivery->tracking_code, trim($data['tracking_code']))) {
            throw ValidationException::withMessages(['tracking_code' => 'Tracking code does not match this transfer.']);
        }
        $scanner->receive((int) $partner->id, (int) $parcelTransfer->to_sorting_center_id,
            (int) $request->user()->id, $data['tracking_code']);

    {
        return Order::with(['seller', 'items.product', 'pickupRequest.originSortingCenter'])

            ->paymentEligible()
            ->whereHas('pickupRequest', function ($query) use ($partner) {
                $query->where('logistics_partner_id', $partner->id)
                    ->where('status', PickupRequest::REQUESTED);
            })
            ->whereNotIn('id', Delivery::select('order_id'))
            ->latest()->get();

            && $coverage->availableForPartner($area, $partner);

    private function validatedAssignment(LogisticsPartner $partner, mixed $centerId, mixed $areaId): array
    {
        $center = null;
        $area = null;

        if ($centerId) {
            $center = SortingCenter::where('logistics_partner_id', $partner->id)
                ->where('status', 'active')
                ->findOrFail((int) $centerId);
        }

        if ($areaId) {
            $area = LogisticsCoverageArea::where('logistics_partner_id', $partner->id)
                ->with('sortingCenter')
                ->findOrFail((int) $areaId);

            if ($area->sorting_center_id) {
                if ($center && (int) $center->id !== (int) $area->sorting_center_id) {
                    throw ValidationException::withMessages([
                        'coverage_area_id' => 'The selected delivery area belongs to a different Sorting Center.',
                    ]);
                }

                $center = $area->sortingCenter;
            }
        }

        if (! $center) {
            $center = $partner->mainSortingCenter()->where('status', 'active')->first()
                ?? $partner->sortingCenters()->where('status', 'active')->first();
        }

        return [$center?->id, $area?->id];
    }

                $approved ? null : $rider->rejection_reason,
            ));
        } catch (\Throwable $exception) {
            // The review decision is authoritative even when outbound mail is temporarily unavailable.
            report($exception);
        }
    }

        if ($ownedCenter->is_main) {
            return back()->withErrors(['sorting_center' => 'The Main Sorting Center must remain active.']);
        }

        $next = $ownedCenter->status === 'active' ? 'inactive' : 'active';
        if ($next === 'inactive') {
            $hasParcels = Delivery::where('logistics_partner_id', $partner->id)
                ->whereNotIn('status', [Delivery::DELIVERED, Delivery::RETURNED])
                ->where(function ($query) use ($ownedCenter) {
                    $query->where('origin_sorting_center_id', $ownedCenter->id)
                        ->orWhere('current_sorting_center_id', $ownedCenter->id)
                        ->orWhere('destination_sorting_center_id', $ownedCenter->id);
                })->exists();
            $hasTransfers = ParcelTransfer::where('status', ParcelTransfer::IN_TRANSIT)
                ->where(function ($query) use ($ownedCenter) {
                    $query->where('from_sorting_center_id', $ownedCenter->id)
                        ->orWhere('to_sorting_center_id', $ownedCenter->id);
                })->exists();
            if ($hasParcels || $hasTransfers) {
                return back()->withErrors(['sorting_center' => 'Branch has unfinished parcels or transfers. Complete or reroute them before deactivation.']);
            }
            // Invalidate staff tokens when branch becomes inactive; re-login after activation.
            foreach ($ownedCenter->staff as $staff) $staff->tokens()->delete();
        } else {
            if (! $ownedCenter->staff()->where('role', 'manager')->where('status', 'active')->exists()) {
                return back()->withErrors(['sorting_center' => 'Assign an active Branch Manager before activating this branch.']);
            }
            if (! $ownedCenter->coverageAreas()->exists()) {
                return back()->withErrors(['coverage' => 'Add at least one delivery coverage area before activating.']);
            }
            $this->assertNoOverlappingCoverage($partner, $ownedCenter);
        }
        $ownedCenter->update([
            'status' => $next,
            'deactivated_at' => $next === 'inactive' ? now() : null,
        ]);

        return redirect()->route('logistics.sorting-centers.show', $ownedCenter)
            ->with('status', 'Branch is now '.($next === 'active' ? 'active and ready for assignments.' : 'inactive.'));
    }

    private function syncCoverage(LogisticsPartner $partner, SortingCenter $center, array $coverage): void
    {
        // Never re-create area IDs when an address-only edit leaves coverage unchanged.
        $normalize = static function (string $type, string $name, string $cities): string {
            $cityList = strtoupper(trim($cities)) === 'ALL' || trim($cities) === ''
                ? ['ALL'] : collect(preg_split('/[|,]+/', $cities))
                    ->map(fn ($v) => mb_strtolower(trim($v)))
                    ->filter()->unique()->sort()->values()->all();
            return mb_strtolower(trim($type)).'|'.mb_strtolower(trim($name)).'|'.implode('|', $cityList);
        };
        $existing = $center->coverageAreas()->get()->map(
            fn ($row) => $normalize($row->area_type, $row->area_name, $row->cities ?: 'ALL')
        )->sort()->values()->all();
        $incoming = collect($coverage)->map(
            fn ($row) => $normalize($row['area_type'], $row['area_name'], (string) ($row['cities'] ?? 'ALL'))
        )->sort()->values()->all();
        if ($existing === $incoming) return;

        $existingIds = $center->coverageAreas()->pluck('id');
        if ($existingIds->isNotEmpty()
            && DB::table('deliveries')->whereIn('destination_area_id', $existingIds)->exists()) {
            throw ValidationException::withMessages([
                'coverage' => 'This coverage is referenced by existing parcels. Keep those areas until the deliveries finish.',
            ]);
        }
        $center->coverageAreas()->delete();

        foreach ($coverage as $row) {
            $cities = trim((string) ($row['cities'] ?? ''));
            $cities = $cities === '' ? 'ALL' : collect(preg_split('/[|,]+/', $cities))
                ->map(fn ($value) => trim((string) $value))
                ->filter()
                ->unique(fn ($value) => strtolower($value))
                ->implode('|');

            $partner->coverageAreas()->create([
                'sorting_center_id' => $center->id,
                'area_name' => trim($row['area_name']),
                'area_type' => $row['area_type'],
                'cities' => $cities ?: 'ALL',
            ]);
        }
    }

    /** Prevent competing active branches in the same province/region and city. */
    private function assertNoOverlappingCoverage(LogisticsPartner $partner, SortingCenter $center): void
    {
        $matcher = app(PhilippineDeliveryCoverage::class);
        $mainActive = $partner->sortingCenters()->where('is_main', true)
            ->where('status', 'active')->exists();
        $others = $partner->coverageAreas()
            ->where(function ($query) use ($center) {
                $query->where('sorting_center_id', '!=', $center->id)
                    ->orWhereNull('sorting_center_id');
            })
            ->with('sortingCenter')->get()
            ->filter(fn ($area) => $area->sorting_center_id === null
                ? ($mainActive && ! $center->is_main)
                : $area->sortingCenter?->status === 'active');
        foreach ($center->coverageAreas as $candidate) {
            foreach ($others as $existing) {
                if ($matcher->overlaps($candidate, $existing)) {
                    throw ValidationException::withMessages(['coverage' =>
                        'Coverage overlaps an active branch in '.$candidate->area_name.'. Adjust the covered cities before activating.']);
                }
            }
        }
    }

    private function nextCode(LogisticsPartner $partner): string
    {
        $prefix = 'SC-'.str_pad((string) $partner->id, 4, '0', STR_PAD_LEFT).'-';
        $sequence = $partner->sortingCenters()->count() + 1;

        do {
            $code = $prefix.str_pad((string) $sequence, 2, '0', STR_PAD_LEFT);
            $sequence++;
        } while ($partner->sortingCenters()->where('code', $code)->exists());

        return $code;
    }

    private function partner(Request $request): LogisticsPartner
    {
        return LogisticsPartner::where('user_id', $request->user()->id)->first()
            ?? abort(403, 'Logistics profile unavailable.');
    }

            ->where('logistics_partner_id', $rider->logistics_partner_id)
            ->whereHas('order', fn ($order) => $order->where('payment_method', 'cod'))
            ->where(function ($query) use ($rider) {

                    ->orWhere(function ($assigned) use ($rider) {

                    });
            })->latest()->get();

        return view('rider.settlements', compact('deliveries'));
    }

    public function remit(Request $request, int $delivery): RedirectResponse
    {
        $rider = $this->eligible($request);

        return DB::transaction(function () use ($rider, $delivery) {
            // The collector and partner are locked from persisted records; browser money fields are ignored.
            $cash = CodSettlement::where('delivery_id', $delivery)
                ->where('collected_by_rider_id', $rider->id)
                ->where('logistics_partner_id', $rider->logistics_partner_id)
                ->lockForUpdate()->firstOrFail();
            $current = Rider::with('partner.user')->whereKey($rider->id)->lockForUpdate()->first();
            abort_unless($current && in_array($current->status, ['active', 'suspended'], true)
                && $current->logistics_partner_id === $cash->logistics_partner_id
                && $current->email && $current->password
                && $current->partner?->user?->status === 'approved', 403);
            if ($cash->status !== CodSettlement::COLLECTED || $cash->remitted_at) {
                return back()->withErrors(['settlement' => 'Remittance already changed. Refresh and review it.']);
            }
            if (CodSettlement::whereKey($cash->id)->where('status', CodSettlement::COLLECTED)
                ->whereNull('remitted_at')->update([
                    'status' => CodSettlement::REMITTED,
                    'remitted_by_rider_id' => $rider->id, 'remitted_at' => now(),
                ]) !== 1) {
                return back()->withErrors(['settlement' => 'Remittance already changed. Refresh and review it.']);
            }

            return redirect()->route('rider.settlements.index')->with('status', 'COD remittance submitted to Logistics.');
        });
    }

    private function eligible(Request $request): Rider
    {
        // Suspended Riders retain only this cash return lane; delivery actions still require active.rider.
        abort_if(Auth::guard('web')->check(), 403);
        $identity = $request->user('rider');
        abort_unless($identity instanceof Rider, 403);
        $rider = $identity->fresh(['partner.user']);
        abort_unless($rider && in_array($rider->status, ['active', 'suspended'], true)
            && $rider->email && $rider->password && $rider->partner?->user?->status === 'approved', 403);

        return $rider;
    }

    public function complete(Request $request, int $delivery): RedirectResponse
    {
        $order = $this->current($request, $delivery)->order;
        if (! $order?->isPaymentEligible()) {
            return $this->stale();
        }

        $request->validate([
            'cash_collected' => $order->payment_method === 'cod' ? ['required', 'accepted'] : ['prohibited'],
            'proof' => ['required', 'file', 'mimes:jpg,jpeg,png,webp', 'max:5120'],
        ]);

        $path = null;
        try {
            return DB::transaction(function () use ($request, $delivery, &$path) {
                $rider = $request->user('rider');
                $owned = $this->current($request, $delivery, true);
                $this->assertActive($rider->id, $owned->logistics_partner_id);

                if ($owned->status !== Delivery::OUT_FOR_DELIVERY
                    || (int) $owned->delivery_rider_id !== (int) $rider->id
                    || $owned->order?->status !== Order::STATUS_TO_RECEIVE
                    || ! $owned->order->isPaymentEligible()
                    || $owned->proof_path || $owned->delivered_at
                    || CodSettlement::where('delivery_id', $owned->id)->exists()) {
                    return $this->stale();
                }

                $path = $request->file('proof')->store('delivery-proofs', 'local');
                if (! $path) {
                    throw ValidationException::withMessages(['proof' => 'Proof could not be stored.']);
                }

                // Rider delivery ends at DELIVERED. Buyer confirmation separately moves the Order to COMPLETED.
                if (Delivery::whereKey($owned->id)->where('status', Delivery::OUT_FOR_DELIVERY)
                    ->where('rider_id', $rider->id)->whereNull('proof_path')
                    ->update([
                        'status' => Delivery::DELIVERED,
                        'delivered_at' => now(),
                        'delivered_rider_id' => $rider->id,
                        'proof_path' => $path,
                    ]) !== 1) {
                    throw ValidationException::withMessages([
                        'delivery' => 'Delivery changed. Refresh and review it.',
                    ]);
                }

                if ($owned->order->payment_method === 'cod') {
                    CodSettlement::create([
                        'order_id' => $owned->order_id,
                        'delivery_id' => $owned->id,
                        'logistics_partner_id' => $owned->logistics_partner_id,
                        'expected_amount' => $owned->order->total_amount,
                        'collected_amount' => $owned->order->total_amount,
                        'status' => CodSettlement::COLLECTED,
                        'collected_by_rider_id' => $rider->id,
                        'collected_at' => now(),
                    ]);
                }

                return back()->with('status', $owned->order->payment_method === 'cod'
                    ? 'Delivery proof saved and COD collection recorded. Waiting for Buyer confirmation.'
                    : 'Delivery proof saved. Waiting for Buyer confirmation.');
            });
        } catch (Throwable $exception) {
            if ($path) {
                Storage::disk('local')->delete($path);
            }
            throw $exception;

<?php

namespace App\Services;

use App\Models\Buyer\Order\Order;
use App\Models\LogisticsCoverageArea;
use App\Models\LogisticsPartner;

/**
 * Single destination/coverage rule shared by Seller pickup partner selection and
 * Logistics sorting. No external geocoding requests or database writes occur here.
 */
class PhilippineDeliveryCoverage
{
    /** @return array{0: ?string, 1: ?string, 2: ?string} Province, municipality/city, region. */
    public function destination(Order $order): array
    {
        $province = trim((string) $order->delivery_province_name);
        $city = trim((string) $order->delivery_municipality_name);
        $region = trim((string) $order->delivery_region_name);

        // The city snapshot is the authoritative structured value. NCR legitimately
        // has no province; never overwrite its city by parsing the address label.
        if ($city !== '') {
            return [$province ?: null, $city, $region ?: null];
        }

        // Legacy orders without structured municipality snapshots. The address
        // label historically ended in `city, province` outside NCR. If a known
        // NCR region survives, the last segment is the city instead.
        $parts = array_values(array_filter(array_map('trim', explode(',', (string) $order->delivery_address))));
        if ($region !== '' && $this->sameRegion($region, 'NCR') && count($parts) >= 2) {
            return [null, end($parts), $region];
        }
        if (count($parts) < 2) {
            return [null, null, $region ?: null];
        }
        return [$province ?: $parts[count($parts) - 1], $parts[count($parts) - 2], $region ?: null];
    }

    public function coversOrder(LogisticsCoverageArea $area, Order $order): bool
    {
        [$province, $city, $region] = $this->destination($order);
        return $city !== null && $this->matches($area, $province, $city, $region);
    }

    public function matches(LogisticsCoverageArea $area, ?string $province, string $city, ?string $region): bool
    {
        if (! in_array($area->area_type, ['province', 'region'], true) || trim($city) === '') {
            return false;
        }

        $cities = array_values(array_filter(array_map('trim', preg_split('/[|,]+/', (string) $area->cities))));
        $allCities = $cities === [] || in_array('all', array_map([$this, 'normalize'], $cities), true);
        if (! $allCities && ! in_array($this->normalizeCity($city), array_map([$this, 'normalizeCity'], $cities), true)) {
            return false;
        }

        if ($area->area_type === 'province') {
            return $province !== null && $this->normalize($area->area_name) === $this->normalize($province);
        }
        // Regions may be saved as "Region IV-A (CALABARZON)" or "CALABARZON".
        // For legacy NCR records, the historical "province" field may hold NCR.
        if ($region !== null && trim($region) !== '') {
            return $this->sameRegion($area->area_name, $region);
        }
        if ($province !== null && $this->sameRegion($area->area_name, $province)) {
            return true;
        }

        // Do not match arbitrary cities to an unrelated region simply because
        // that region enumerates them. Without region metadata, this is ambiguous
        // (e.g. the same city name occurs in several provinces/regions).
        return false;
    }

    /** Check same-level coverage overlap for branch activation. Cross-level region/province
     *  conflicts still require a real PSGC mapping and are flagged for manual review. */
    public function overlaps(LogisticsCoverageArea $left, LogisticsCoverageArea $right): bool
    {
        if ($left->area_type !== $right->area_type) return false;
        if ($left->area_type === 'region') {
            if (! $this->sameRegion($left->area_name, $right->area_name)) return false;
        } elseif ($left->area_type === 'province') {
            if ($this->normalize($left->area_name) !== $this->normalize($right->area_name)) return false;
        } else {
            return false;
        }
        $a = array_values(array_filter(array_map('trim', preg_split('/[|,]+/', (string) $left->cities))));
        $b = array_values(array_filter(array_map('trim', preg_split('/[|,]+/', (string) $right->cities))));
        if ($a === [] || $b === [] || in_array('all', array_map([$this, 'normalize'], $a), true)
            || in_array('all', array_map([$this, 'normalize'], $b), true)) return true;
        return array_intersect(array_map([$this, 'normalizeCity'], $a),
            array_map([$this, 'normalizeCity'], $b)) !== [];
    }

    public function availableForPartner(LogisticsCoverageArea $area, LogisticsPartner $partner): bool
    {
        if ((int) $area->logistics_partner_id !== (int) $partner->id) {
            return false;
        }
        if ($area->sorting_center_id !== null) {
            return $area->sortingCenter?->status === 'active'
                && (int) $area->sortingCenter->logistics_partner_id === (int) $partner->id;
        }
        // Pre-multi-branch coverage may be unassigned; it belongs to the main hub
        // only if that hub is operating.
        return $partner->sortingCenters()->where('is_main', true)->where('status', 'active')->exists();
    }

    public function sameRegion(?string $a, ?string $b): bool
    {
        $left = $this->regionAliases($a);
        $right = $this->regionAliases($b);
        return $left !== [] && $right !== [] && array_intersect($left, $right) !== [];
    }

    private function regionAliases(?string $value): array
    {
        $raw = trim((string) $value);
        if ($raw === '') return [];
        if (preg_match('/\bncr\b|national capital region|metro manila/i', $raw)) {
            return ['ncr'];
        }
        if (preg_match('/\bnir\b|negros island region/i', $raw)) {
            return ['nir'];
        }
        $aliases = [$this->normalize($raw)];
        if (preg_match('/^(.*?)\s*\(([^()]+)\)\s*$/', $raw, $groups)) {
            $aliases[] = $this->normalize($groups[1]);
            $aliases[] = $this->normalize($groups[2]);
        }
        return array_values(array_unique(array_filter($aliases)));
    }

    private function normalize(string $value): string
    {
        return strtolower(trim(preg_replace('/\s+/u', ' ', $value)));
    }

    private function normalizeCity(string $value): string
    {
        $value = $this->normalize($value);
        $value = preg_replace('/^(?:city|municipality) of\s+/i', '', $value);
        return trim(preg_replace('/\s+city$/i', '', $value));
    }
}

<?php
namespace App\Services;

use App\Models\Buyer\Order\Order;
use App\Models\Finance\OrderFinancial;
use App\Models\Finance\SettlementEntry;
use App\Models\Logistics\CodSettlement;
use App\Models\Logistics\Delivery;
use Illuminate\Support\Facades\DB;
use Illuminate\Validation\ValidationException;

class ShopHopFinance
{
    /** Pure ledger calculation. ShopHop funds platform discounts; seller/logistics remain whole. */
    public static function breakdown(
        float $merchandise, float $sellerDiscount, float $platformDiscount,
        float $shippingFee, float $shippingDiscount, float $codFee
    ): array {
        $sellerBase = max(0, round($merchandise - $sellerDiscount, 2));
        $commission = round($sellerBase * ((float) config('shophop_finance.commission_percent', 10) / 100), 2);
        return [
            'commission' => $commission,
            'seller_earnings' => round($sellerBase - $commission, 2),
            'logistics_earnings' => round($shippingFee, 2),
            'platform_net' => round($commission + $codFee - $platformDiscount - $shippingDiscount, 2),
        ];
    }

    /** Amounts are snapped at checkout, independent of future voucher/rate updates. */
    public function snapshot(Order $order, float $merchandise, float $sellerDiscount, float $platformDiscount, float $shippingDiscount): void
    {
        $calculated = self::breakdown($merchandise, $sellerDiscount, $platformDiscount,
            (float) $order->shipping_fee, $shippingDiscount, (float) $order->cod_fee);
        OrderFinancial::create([
            'order_id' => $order->id, 'merchandise' => round($merchandise, 2),
            'seller_discount' => round($sellerDiscount, 2), 'platform_discount' => round($platformDiscount, 2),
            'shipping_fee' => (float) $order->shipping_fee, 'shipping_discount' => round($shippingDiscount, 2),
            'cod_fee' => (float) $order->cod_fee,
            ...$calculated,
        ]);
    }

    /** Called ONLY by explicit admin action; idempotent under locks and DB unique(order,kind). */
    public function prepareForOrder(int $orderId): void
    {
        DB::transaction(function () use ($orderId) {
            $order = Order::whereKey($orderId)->lockForUpdate()->firstOrFail();
            $financial = OrderFinancial::where('order_id', $order->id)->lockForUpdate()->first();
            if (! $financial) {
                throw ValidationException::withMessages(['order' => 'This Order predates financial snapshots; manual reconciliation required.']);
            }
            if (\App\Models\Logistics\ReturnCase::where('order_id',$order->id)->exists()) {
                throw ValidationException::withMessages(['order' => 'Return case exists; settlement is blocked.']);
            }
            if ($financial->settlement_hold) {
                throw ValidationException::withMessages(['order' => 'This order has an unresolved settlement hold.']);
            }
            if ($order->status !== Order::STATUS_COMPLETED || ! $order->isPaymentEligible()) {
                throw ValidationException::withMessages(['order' => 'Order must be completed and payment eligible first.']);
            }
            $delivery = Delivery::where('order_id', $order->id)->lockForUpdate()->first();
            if (! $delivery || $delivery->status !== Delivery::DELIVERED || ! $delivery->logistics_partner_id) {
                throw ValidationException::withMessages(['order' => 'Verified delivered logistics assignment required.']);
            }
            if ($order->payment_method === 'cod') {
                $cod = CodSettlement::where('order_id', $order->id)->lockForUpdate()->first();
                if (! $cod || $cod->status !== CodSettlement::RECONCILED
                    || ! $cod->admin_remitted_at || ! $cod->admin_received_at
                    || abs((float) $cod->received_amount - (float) $cod->expected_amount) > 0.001
                    || abs((float) $cod->admin_received_amount - (float) $cod->expected_amount) > 0.001) {
                    throw ValidationException::withMessages(['order' => 'COD must be fully reconciled before releasing payouts.']);
                }
            }
            foreach ([
                ['seller', (int) $order->seller_id, (float) $financial->seller_earnings],
                ['logistics', (int) $delivery->logistics_partner_id, (float) $financial->logistics_earnings],
            ] as [$kind, $payee, $amount]) {
                if ($payee <= 0 || $amount < 0) throw ValidationException::withMessages(['order' => 'Invalid settlement beneficiary.']);
                SettlementEntry::firstOrCreate(['order_id' => $orderId, 'kind' => $kind],
                    ['payee_id' => $payee, 'amount' => round($amount, 2), 'status' => 'pending']);
            }
            // Rider fees are funded from logistics revenue, not the ShopHop product commission.
            // Only assign if the actual rider/role is recorded. Each fee is capped by remaining shipping revenue.
            $remaining = (float) $financial->logistics_earnings;
            foreach ([
                ['pickup_rider', (int) $delivery->pickup_rider_id, (float) config('shophop_finance.pickup_rider_fee', 20)],
                ['delivery_rider', (int) ($delivery->delivered_rider_id ?: $delivery->delivery_rider_id), (float) config('shophop_finance.delivery_rider_fee', 30)],
            ] as [$kind, $riderId, $rate]) {
                if ($riderId <= 0) continue;
                $amount = round(min($remaining, max(0, $rate)), 2);
                $remaining -= $amount;
                SettlementEntry::firstOrCreate(['order_id' => $orderId, 'kind' => $kind],
                    ['payee_id' => $riderId, 'amount' => $amount, 'status' => 'pending']);
            }
            // Transit Rider is paid only for a real, received transfer. Multiple legs have
            // separate unique kind keys. The combined Rider costs never exceed shipping revenue.
            $receivedTransfers = $delivery->transfers()->where('status', \App\Models\Logistics\ParcelTransfer::RECEIVED)
                ->whereNotNull('transit_rider_id')->orderBy('id')->get();
            foreach ($receivedTransfers as $transfer) {
                if ($remaining <= 0) break;
                $amount = round(min($remaining, max(0, (float) config('shophop_finance.transit_rider_fee', 40))), 2);
                $remaining = max(0, round($remaining - $amount, 2));
                SettlementEntry::firstOrCreate(['order_id' => $orderId, 'kind' => 'transit_'.$transfer->id], [
                    'payee_id' => (int) $transfer->transit_rider_id,
                    'amount' => $amount,
                    'status' => 'pending',
                ]);
            }
        });
    }

    /** Admin marks seller/logistics money manually transferred; no wallet transfer occurs. */
    public function markAdminPaid(int $entryId, int $adminId, string $reference): void
    {
        DB::transaction(function () use ($entryId, $adminId, $reference) {
            $candidate = SettlementEntry::whereKey($entryId)->firstOrFail();
            $order = Order::whereKey($candidate->order_id)->lockForUpdate()->firstOrFail();
            $entry = SettlementEntry::whereKey($entryId)->lockForUpdate()->firstOrFail();
            if ($order->status !== Order::STATUS_COMPLETED || ! $order->isPaymentEligible()) {
                throw ValidationException::withMessages(['settlement' => 'Order is no longer payment/receipt eligible.']);
            }
            if ($order->payment_method === 'cod') {
                $cod = CodSettlement::where('order_id', $order->id)->lockForUpdate()->first();
                if (! $cod || ! $cod->admin_received_at
                    || abs((float) $cod->admin_received_amount - (float) $cod->expected_amount) > .001) {
                    throw ValidationException::withMessages(['settlement' => 'ShopHop Admin must confirm receipt of COD cash before payouts.']);
                }
            }
            if (! in_array($entry->kind, ['seller', 'logistics'], true) || $entry->status !== 'pending') {
                throw ValidationException::withMessages(['settlement' => 'Only pending Seller/Logistics payouts can be recorded.']);
            }
            $financial = OrderFinancial::where('order_id', $entry->order_id)->lockForUpdate()->firstOrFail();
            if ($financial->settlement_hold) throw ValidationException::withMessages(['settlement' => 'Order is on hold.']);
            $entry->update(['status' => 'paid', 'payment_reference' => $reference, 'paid_by_user_id' => $adminId, 'paid_at' => now()]);
        });
    }

    /** A Logistics partner can only mark its own Rider's fee after its own settlement is paid. */
    public function markRiderPaid(int $entryId, int $partnerId, int $operatorId, string $reference): void
    {
        DB::transaction(function () use ($entryId, $partnerId, $operatorId, $reference) {
            $entry = SettlementEntry::whereKey($entryId)->lockForUpdate()->firstOrFail();
            if (! (in_array($entry->kind, ['pickup_rider', 'delivery_rider'], true)
                || str_starts_with($entry->kind, 'transit_')) || $entry->status !== 'pending') {
                throw ValidationException::withMessages(['settlement' => 'Rider fee already settled or invalid.']);
            }
            $logistics = SettlementEntry::where('order_id', $entry->order_id)
                ->where('kind', 'logistics')->lockForUpdate()->first();
            if (! $logistics || (int) $logistics->payee_id !== $partnerId || $logistics->status !== 'paid') {
                throw ValidationException::withMessages(['settlement' => 'Logistics payout must be recorded as paid before rider settlement.']);
            }
            if (str_starts_with($entry->kind, 'transit_')) {
                $transferId = (int) substr($entry->kind, strlen('transit_'));
                if (! \App\Models\Logistics\ParcelTransfer::whereKey($transferId)
                    ->whereHas('delivery', fn ($query) => $query->where('order_id', $entry->order_id)->where('logistics_partner_id', $partnerId))
                    ->where('transit_rider_id', $entry->payee_id)
                    ->where('status', \App\Models\Logistics\ParcelTransfer::RECEIVED)->exists()) {
                    throw ValidationException::withMessages(['settlement' => 'Transit Rider/transfer not eligible for payout.']);
                }
            }
            $entry->update(['status' => 'paid', 'payment_reference' => $reference, 'paid_at' => now(), 'paid_by_user_id' => $operatorId]);
        });
    }
}

<script>
document.addEventListener('DOMContentLoaded', function () {

    // COD is the only payment method: the handling fee always applies, once per shop.
    const codFeePerShop = {{ (float) $codFee }};

    const voucherCatalog = @json(
        $availableVouchers->mapWithKeys(fn ($voucher) => [$voucher['code'] => $voucher])
    );

    const initialVoucherCode = @json($initialVoucherCode);
    const platformCatalog = @json(collect($platformVouchers)->keyBy('code'));
    const platformInput = document.getElementById('platformVoucherInput');
    const shippingInput = document.getElementById('shippingVoucherInput');
    const platformFeedback = document.getElementById('platformVoucherFeedback');


    const shopGroups = document.querySelectorAll('.checkout-shop-group');

    const summarySubtotal = document.getElementById('summarySubtotal');
    const summaryShipping = document.getElementById('summaryShipping');
    const summaryProductSavings = document.getElementById('summaryProductSavings');

    const summaryVoucherRow = document.getElementById('summaryVoucherRow');
    const summaryVoucherCode = document.getElementById('summaryVoucherCode');
    const summaryVoucherDiscount = document.getElementById('summaryVoucherDiscount');

    const summaryCodFee = document.getElementById('summaryCodFee');
    const summaryTotal = document.getElementById('summaryTotal');
    const summarySavedText = document.getElementById('summarySavedText');

    const mobileSummaryTotal = document.getElementById('mobileSummaryTotal');

    const voucherInput = document.getElementById('voucherInput');
    const applyVoucherBtn = document.getElementById('applyVoucherBtn');
    const voucherMessage = document.getElementById('voucherMessage');
    const voucherAppliedBadge = document.getElementById('voucherAppliedBadge');
    const voucherCodeHidden = document.getElementById('voucherCodeHidden');

    const checkoutForm = document.getElementById('checkoutForm');
    const mobilePlaceOrderBtn = document.getElementById('mobilePlaceOrderBtn');

    const checkoutToast = document.getElementById('checkoutToast');

    let activeVoucherCode =
        initialVoucherCode && voucherCatalog[initialVoucherCode]
            ? initialVoucherCode
            : '';

    let currentVoucherDiscount = 0;


    function formatPeso(amount) {
        // Summary previews retain cents so the server's persisted decimal total is not visually rounded.
        return '₱' + Math.max(0, Number(amount) || 0)
            .toLocaleString('en-PH', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
    }


    function showToast(message) {
        if (!checkoutToast) return;

        checkoutToast.textContent = message;

        checkoutToast.classList.remove('translate-y-6', 'opacity-0');
        checkoutToast.classList.add('translate-y-0', 'opacity-100');

        clearTimeout(showToast.timer);

        showToast.timer = setTimeout(function () {
            checkoutToast.classList.remove('translate-y-0', 'opacity-100');
            checkoutToast.classList.add('translate-y-6', 'opacity-0');
        }, 2200);
    }


    function voucherDiscountForShop(shopSubtotal) {
        const voucher = activeVoucherCode ? voucherCatalog[activeVoucherCode] : null;

        if (!voucher) return 0;

        const minimum = Number(voucher.min_spend || 0);

        if (shopSubtotal < minimum) {
            return 0;
        }

        if (voucher.type === 'percent') {
            return shopSubtotal * (Number(voucher.value || 0) / 100);
        }

        return Math.min(Number(voucher.value || 0), shopSubtotal);
    }


    function syncVoucherUI(voucherEligibleShopSubtotal, discount) {
        const voucher = activeVoucherCode ? voucherCatalog[activeVoucherCode] : null;

        currentVoucherDiscount = discount;

        if (voucherCodeHidden) {
            voucherCodeHidden.value = voucher ? activeVoucherCode : '';
        }

        if (voucherAppliedBadge) {
            voucherAppliedBadge.classList.toggle('hidden', !voucher);
        }

        if (summaryVoucherCode) {
            summaryVoucherCode.textContent =
                voucher ? '(' + activeVoucherCode + ')' : '';
        }

        if (summaryVoucherDiscount) {
            summaryVoucherDiscount.textContent =
                '-' + formatPeso(currentVoucherDiscount);
        }

        if (summaryVoucherRow) {
            const show = voucher && currentVoucherDiscount > 0;

            summaryVoucherRow.classList.toggle('hidden', !show);
            summaryVoucherRow.classList.toggle('flex', !!show);
        }

        if (voucher) {
            const minimum = Number(voucher.min_spend || 0);

            if (voucherEligibleShopSubtotal >= minimum) {
                voucherMessage.textContent =
                    activeVoucherCode + ' applied — you save ' +
                    formatPeso(currentVoucherDiscount) + '.';

                voucherMessage.classList.remove('hidden', 'text-red-500');
                voucherMessage.classList.add('text-teal-dark');
            } else {
                voucherMessage.textContent =
                    'Spend ' + formatPeso(minimum) +
                    ' on this seller\'s items to use ' + activeVoucherCode + '.';

                voucherMessage.classList.remove('hidden', 'text-teal-dark');
                voucherMessage.classList.add('text-red-500');
            }
        }
    }


    function promoDiscount(input, kind, eligible, merchandiseBase) {
        const code = (input?.value || '').trim().toUpperCase();
        if (!code) return 0;

        const promo = platformCatalog[code];

        if (!promo || promo.kind !== kind || merchandiseBase < Number(promo.min_spend || 0) || eligible <= 0) {
            return 0;
        }

        return Math.round(Math.min(eligible, promo.type === 'percent'
            ? eligible * Number(promo.value) / 100
            : Number(promo.value)) * 100) / 100;
    }


    function recalculate() {
        let merchandiseSubtotal = 0;
        let shippingTotal = 0;
        let productSavings = 0;
        let voucherEligibleShopSubtotal = 0;
        let voucherDiscount = 0;

        const voucher = activeVoucherCode ? voucherCatalog[activeVoucherCode] : null;

        shopGroups.forEach(function (group, index) {
            let groupSubtotal = 0;
            let eligibleSubtotal = 0;

            group.querySelectorAll('.checkout-item').forEach(function (row) {
                const price = Number(row.dataset.price || 0);
                const originalPrice = Number(row.dataset.originalPrice || price);

                const qtyInput = row.querySelector('[data-qty-input]');
                const qty = Math.max(1, parseInt(qtyInput?.value, 10) || 1);

                groupSubtotal += price * qty;

                // Match the server voucher rule by counting only assigned Product rows in this seller group.
                if (voucher && String(group.dataset.shopId) === String(voucher.seller_id)
                    && voucher.product_ids.includes(Number(row.dataset.productId))) {
                    eligibleSubtotal += price * qty;
                }

                productSavings += Math.max(0, originalPrice - price) * qty;
            });

            merchandiseSubtotal += groupSubtotal;

            const selectedShipping = group.querySelector(
                'input[type="radio"][name^="shipping_method"]:checked'
            );

            shippingTotal += Number(selectedShipping?.dataset.fee || 0);

            const groupSubtotalEl = group.querySelector(
                '[data-shop-subtotal="' + index + '"]'
            );

            if (groupSubtotalEl) {
                groupSubtotalEl.textContent = formatPeso(groupSubtotal);
            }

            if (voucher && String(group.dataset.shopId) === String(voucher.seller_id)) {
                voucherEligibleShopSubtotal = eligibleSubtotal;
                voucherDiscount = voucherDiscountForShop(eligibleSubtotal);
            }
        });

        syncVoucherUI(voucherEligibleShopSubtotal, voucherDiscount);

        const codFeeTotal = codFeePerShop * shopGroups.length;

        const netMerchandise = Math.max(0, merchandiseSubtotal - currentVoucherDiscount);
        const platformSaving = promoDiscount(platformInput, 'merchandise', netMerchandise, netMerchandise);
        const shippingSaving = promoDiscount(shippingInput, 'shipping', shippingTotal, netMerchandise);

        const platformRow = document.getElementById('summaryPlatformRow');
        const shippingRow = document.getElementById('summaryShippingPromoRow');

        platformRow.classList.toggle('hidden', platformSaving <= 0);
        platformRow.classList.toggle('flex', platformSaving > 0);
        shippingRow.classList.toggle('hidden', shippingSaving <= 0);
        shippingRow.classList.toggle('flex', shippingSaving > 0);

        document.getElementById('summaryPlatformDiscount').textContent = '-' + formatPeso(platformSaving);
        document.getElementById('summaryShippingPromoDiscount').textContent = '-' + formatPeso(shippingSaving);

        if (platformFeedback) {
            platformFeedback.textContent = 'ShopHop-funded savings: ' + formatPeso(platformSaving + shippingSaving) +
                '. Final eligibility and amount are validated by the server.';
        }

        const total = Math.max(0, merchandiseSubtotal + shippingTotal + codFeeTotal
            - currentVoucherDiscount - platformSaving - shippingSaving);

        summarySubtotal.textContent = formatPeso(merchandiseSubtotal);
        summaryShipping.textContent = formatPeso(shippingTotal);
        summaryProductSavings.textContent = '-' + formatPeso(productSavings);
        summaryCodFee.textContent = formatPeso(codFeeTotal);

        summaryTotal.textContent = formatPeso(total);
        mobileSummaryTotal.textContent = formatPeso(total);

        summarySavedText.textContent =
            'You save ' + formatPeso(productSavings + currentVoucherDiscount + platformSaving + shippingSaving);
    }


    [platformInput, shippingInput].forEach(function (input) {
        input?.addEventListener('input', recalculate);
        input?.addEventListener('change', recalculate);
    });


    // Tap a ShopHop code chip to fill the matching input.
    document.querySelectorAll('.platform-suggestion').forEach(function (button) {
        button.addEventListener('click', function () {
            const target = button.dataset.kind === 'shipping' ? shippingInput : platformInput;

            if (!target) return;

            target.value = button.dataset.code;
            recalculate();
            showToast(button.dataset.code + ' added.');
        });
    });


    /*
    |--------------------------------------------------------------------------
    | Quantity
    |--------------------------------------------------------------------------
    */
    document.querySelectorAll('.checkout-item').forEach(function (row) {

        const decreaseButton = row.querySelector('[data-qty-decrease]');
        const increaseButton = row.querySelector('[data-qty-increase]');
        const qtyInput = row.querySelector('[data-qty-input]');

        if (!decreaseButton || !increaseButton || !qtyInput) {
            return;
        }

        const max = parseInt(qtyInput.max, 10) || 1;


        function normalizeQty(value) {
            return Math.max(1, Math.min(max, parseInt(value, 10) || 1));
        }


        decreaseButton.addEventListener('click', function () {
            qtyInput.value = normalizeQty((parseInt(qtyInput.value, 10) || 1) - 1);
            recalculate();
        });


        increaseButton.addEventListener('click', function () {
            qtyInput.value = normalizeQty((parseInt(qtyInput.value, 10) || 1) + 1);
            recalculate();
        });


        qtyInput.addEventListener('change', function () {
            qtyInput.value = normalizeQty(qtyInput.value);
            recalculate();
        });

    });


    /*
    |--------------------------------------------------------------------------
    | Shipping option
    |--------------------------------------------------------------------------
    */
    document.querySelectorAll('[data-shipping-option]').forEach(function (option) {

        const radio = option.querySelector('input[type="radio"]');

        option.addEventListener('click', function () {
            radio.checked = true;

            const group = option.closest('[data-shipping-group]');

            group.querySelectorAll('[data-shipping-option]').forEach(function (item) {
                item.classList.remove('border-teal', 'bg-teal-light/35');
                item.classList.add('border-gray-border');
            });

            option.classList.remove('border-gray-border');
            option.classList.add('border-teal', 'bg-teal-light/35');

            recalculate();
        });

    });


    /*
    |--------------------------------------------------------------------------
    | Voucher
    |--------------------------------------------------------------------------
    */
    function applyVoucher(code, showFeedback = true) {
        const normalized = String(code || '').trim().toUpperCase();

        voucherInput.value = normalized;

        if (!normalized || !voucherCatalog[normalized]) {
            activeVoucherCode = '';

            voucherMessage.textContent =
                normalized ? 'Invalid or expired voucher code.' : 'Enter a voucher code first.';

            voucherMessage.classList.remove('hidden', 'text-teal-dark');
            voucherMessage.classList.add('text-red-500');

            recalculate();

            if (showFeedback && normalized) {
                showToast('Voucher not available.');
            }

            return;
        }

        // A voucher code is eligible only when an assigned Product is selected from its seller.
        const belongsToCart = Array.from(shopGroups).some((group) =>
            String(group.dataset.shopId) === String(voucherCatalog[normalized].seller_id)
            && Array.from(group.querySelectorAll('.checkout-item')).some((row) =>
                voucherCatalog[normalized].product_ids.includes(Number(row.dataset.productId))
            )
        );

        if (!belongsToCart) {
            activeVoucherCode = '';

            voucherMessage.textContent = 'This voucher does not apply to any seller in your order.';
            voucherMessage.classList.remove('hidden', 'text-teal-dark');
            voucherMessage.classList.add('text-red-500');

            recalculate();

            if (showFeedback) {
                showToast('Voucher not applicable.');
            }

            return;
        }

        activeVoucherCode = normalized;

        recalculate();

        if (showFeedback) {
            // The server applies the code only after final eligibility and price validation.
            showToast(currentVoucherDiscount > 0
                ? normalized + ' selected for Checkout.'
                : 'This voucher does not meet the eligible spend minimum.');
        }
    }

        applyVoucher(voucherInput.value);
    });

        if (event.key === 'Enter') {
            event.preventDefault();
            applyVoucher(voucherInput.value);
        }
    });

            applyVoucher(button.dataset.code);

        applyVoucher(initialVoucherCode, false);

        recalculate();
    }

});
