<?php

namespace App\Http\Controllers\Branch;

use App\Http\Controllers\Controller;
use App\Models\Logistics\Delivery;
use App\Models\Logistics\ParcelTransfer;
use App\Models\Logistics\PickupRequest;
use App\Models\Logistics\SortingCenter;
use Illuminate\Http\Request;
use Illuminate\View\View;

class DashboardController extends Controller
{
    public function index(Request $request): View
    {
        $center = SortingCenter::where('user_id', $request->user()->id)
            ->where('status', 'active')->whereNull('archived_at')->first();
        abort_unless($center, 403, 'Walang aktibong branch na naka-link sa account mo.');

        // Papunta sa branch: (a) pickup na dala ng rider, (b) transfers na umalis na
        $pickedUp = Delivery::with('order.seller')
            ->where('origin_sorting_center_id', $center->id)
            ->where('status', Delivery::PICKED_UP);

        $transfersIn = ParcelTransfer::with(['delivery', 'fromCenter'])
            ->where('to_sorting_center_id', $center->id)
            ->where('status', ParcelTransfer::IN_TRANSIT)
            ->where(fn ($q) => $q->whereNull('transit_rider_id')->orWhereNotNull('departed_at'));

        $awaiting = (clone $pickedUp)->latest('picked_up_at')->limit(10)->get()->map(fn ($d) => [
            'tracking' => $d->tracking_code,
            'label'    => 'Pickup mula sa seller',
            'from'     => $d->order?->seller?->business_name ?? 'Seller',
            'when'     => $d->picked_up_at?->diffForHumans() ?? 'â€”',
        ])->concat(
            (clone $transfersIn)->latest('dispatched_at')->limit(10)->get()->map(fn ($t) => [
                'tracking' => $t->delivery?->tracking_code,
                'label'    => 'Transfer',
                'from'     => $t->fromCenter?->name ?? 'Hub',
                'when'     => $t->dispatched_at?->diffForHumans() ?? 'â€”',
            ])
        )->filter(fn ($a) => $a['tracking'])->values();

        $pendingPickups = PickupRequest::where('origin_sorting_center_id', $center->id)
            ->where('status', PickupRequest::REQUESTED)->count();

        $atBranch = Delivery::where('current_sorting_center_id', $center->id)
            ->whereIn('status', [Delivery::AT_SORTING_CENTER, Delivery::SORTED])->count();

        $toAssignQuery = Delivery::with('order')
            ->where('destination_sorting_center_id', $center->id)
            ->where('current_sorting_center_id', $center->id)
            ->whereIn('status', [Delivery::SORTED, Delivery::DELIVERY_FAILED]);

        $outForDelivery = Delivery::where('destination_sorting_center_id', $center->id)
            ->whereIn('status', [Delivery::DELIVERY_ASSIGNED, Delivery::OUT_FOR_DELIVERY])->count();

        $deliveredToday = Delivery::where('destination_sorting_center_id', $center->id)
            ->where('status', Delivery::DELIVERED)->whereDate('delivered_at', today())->count();

        $failed = Delivery::where('destination_sorting_center_id', $center->id)
            ->where('status', Delivery::DELIVERY_FAILED)->count();

        $stats = [
            'pickups'          => $pendingPickups,
            'to_receive'       => (clone $pickedUp)->count() + (clone $transfersIn)->count(),
            'at_branch'        => $atBranch,
            'to_assign'        => (clone $toAssignQuery)->count(),
            'out_for_delivery' => $outForDelivery,
            'delivered_today'  => $deliveredToday,
            'failed'           => $failed,
            'cod_with_riders'  => 0, // saka na kapag ginawa na ang rider/COD part
        ];

        $toAssign = $toAssignQuery->latest('updated_at')->limit(5)->get()->map(fn ($d) => [
            'tracking' => $d->tracking_code,
            'receiver' => $d->order?->delivery_name,
            'barangay' => $d->order?->delivery_barangay_name ?: $d->order?->delivery_address,
            'cod'      => $d->order && $d->order->payment_method === 'cod' ? (float) $d->order->total_amount : null,
        ]);

        $riders = []; // saka na

        $alerts = [];
        if ($failed > 0) {
            $alerts[] = ['level' => 'warning', 'title' => "{$failed} parcel(s) failed delivery", 'detail' => 'Kailangan ng retry o return.'];
        }
        if ($pendingPickups > 0) {
            $alerts[] = ['level' => 'info', 'title' => "{$pendingPickups} pickup request(s) naghihintay ng rider", 'detail' => 'I-assign sa Logistics board.'];
        }

        return view('branch.dashboard.index', compact('center', 'stats', 'awaiting', 'toAssign', 'riders', 'alerts'));
    }
}

<?php

// app/Http/Controllers/branch/ParcelController.php

namespace App\Http\Controllers\branch;

use App\Http\Controllers\Controller;
use App\Models\Logistics\Delivery;
use App\Models\Logistics\Rider;
use App\Models\Logistics\SortingCenter;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\View\View;

class ParcelController extends Controller
{
    /**
     * Tab key sa view => status values sa deliveries table.
     * ASSUMPTION: i-adjust sa totoong status values ng Delivery model mo.
     */
    private const STATUS_MAP = [
        'at_branch'        => ['at_branch', 'sorted'],
        'out_for_delivery' => ['out_for_delivery'],
        'delivered'        => ['delivered'],
        'failed'           => ['failed', 'failed_attempt'],
        'returning'        => ['returning', 'return_to_sender'],
    ];

    /** Mga status na pwedeng i-assign sa rider. */
    private const ASSIGNABLE = ['at_branch', 'sorted', 'failed', 'failed_attempt'];

    public function index(Request $request): View
    {
        $center = $this->currentCenter($request);

        $status = $request->query('status', 'all');
        if ($status !== 'all' && ! array_key_exists($status, self::STATUS_MAP)) {
            $status = 'all';
        }
        $search = trim((string) $request->query('q', ''));

        $base = Delivery::query()->where('sorting_center_id', $center->id);

        // Stats (isang query, hindi apektado ng filter/search)
        $rows = (clone $base)->select('status', DB::raw('COUNT(*) as total'))
            ->groupBy('status')->pluck('total', 'status');

        $countFor = fn (string $key) => collect(self::STATUS_MAP[$key])
            ->sum(fn ($s) => (int) ($rows[$s] ?? 0));

        $counts = [
            'at_branch'        => $countFor('at_branch'),
            'out_for_delivery' => $countFor('out_for_delivery'),
            'failed'           => $countFor('failed'),
        ];

        $parcels = $base
            ->with(['order', 'rider'])
            ->when($status !== 'all', fn ($q) => $q->whereIn('status', self::STATUS_MAP[$status]))
            ->when($search !== '', function ($q) use ($search) {
                $like = '%'.str_replace(['%', '_'], ['\%', '\_'], $search).'%';
                $q->where(function ($w) use ($like) {
                    $w->where('tracking_number', 'like', $like)
                      ->orWhere('receiver_name', 'like', $like)
                      ->orWhereHas('order', fn ($o) => $o->where('order_number', 'like', $like));
                });
            })
            ->latest('updated_at')
            ->paginate(20)
            ->withQueryString()
            ->through(fn (Delivery $d) => $this->present($d));

        $riders = Rider::query()
            ->where('sorting_center_id', $center->id)
            ->where('status', 'active')
            ->orderBy('name')
            ->get(['id', 'name']);

        return view('branch.parcels.index', [
            'center'       => $center,
            'parcels'      => $parcels,
            'counts'       => $counts,
            'riders'       => $riders,
            'activeStatus' => $status,
            'search'       => $search,
        ]);
    }

    public function assign(Request $request): RedirectResponse
    {
        $center = $this->currentCenter($request);

        $data = $request->validate([
            'parcel_ids'   => ['required', 'array', 'min:1', 'max:100'],
            'parcel_ids.*' => ['string', 'max:64'],
            'rider_id'     => ['required', 'integer'],
        ], [
            'parcel_ids.required' => 'Pumili muna ng kahit isang parcel.',
            'rider_id.required'   => 'Pumili ng rider.',
        ]);

        // Dapat rider ng branch na ito at active
        $rider = Rider::query()
            ->where('sorting_center_id', $center->id)
            ->where('status', 'active')
            ->find($data['rider_id']);

        if (! $rider) {
            return back()->with('error', 'Hindi valid ang rider na napili.');
        }

        $assigned = DB::transaction(function () use ($center, $rider, $data) {
            $deliveries = Delivery::query()
                ->where('sorting_center_id', $center->id)
                ->whereIn('tracking_number', $data['parcel_ids'])
                ->whereIn('status', self::ASSIGNABLE)
                ->lockForUpdate()
                ->get();

            foreach ($deliveries as $delivery) {
                $delivery->update([
                    'rider_id' => $rider->id,
                    'status'   => 'out_for_delivery',
                ]);
            }

            return $deliveries->count();
        });

        if ($assigned === 0) {
            return back()->with('error', 'Walang na-assign. Baka na-update na ang mga parcel.');
        }

        $skipped = count($data['parcel_ids']) - $assigned;
        $message = "{$assigned} parcel(s) na-assign kay {$rider->name}.";
        if ($skipped > 0) {
            $message .= " {$skipped} ang nilaktawan dahil hindi na pwedeng i-assign.";
        }

        return back()->with('success', $message);
    }

    /** Hugis ng data na hinihingi ng branch.parcels.index view. */
    private function present(Delivery $d): array
    {
        $viewStatus = collect(self::STATUS_MAP)
            ->search(fn ($values) => in_array($d->status, $values, true)) ?: 'at_branch';

        return [
            'tracking' => $d->tracking_number,
            'order'    => $d->order->order_number ?? 'â€”',
            'receiver' => $d->receiver_name,
            'barangay' => $d->delivery_address,
            'cod'      => (float) ($d->cod_amount ?? 0),
            'status'   => $viewStatus,
            'rider'    => $d->rider->name ?? null,
            'updated'  => optional($d->updated_at)->format('M d, g:i A'),
            'note'     => $d->failure_note ?? null,
        ];
    }

    /**
     * ASSUMPTION: paano malaman ang branch ng naka-login.
     * Palitan ayon sa auth setup mo (guard, relation, o session).
     */
    private function currentCenter(Request $request): SortingCenter
    {
        $user = $request->user();

        $center = SortingCenter::query()
            ->where('id', $user->sorting_center_id ?? null)
            ->first();

        abort_unless($center, 403, 'Walang naka-link na branch sa account mo.');

        return $center;
    }
}

    }

    /** Assign the rider who will collect the parcel from the Seller and bring it to the sorting center. */
    public function assign(Request $request, int $order): RedirectResponse
    {
        $partner = $this->partner($request);
        $data = $request->validate(['rider_id' => ['required', 'integer']]);
        $rider = $this->activeRider($partner, (int) $data['rider_id']);

        try {
            return DB::transaction(function () use ($partner, $rider, $order) {
                $ready = Order::with('pickupRequest')->whereKey($order)->lockForUpdate()->firstOrFail();
                $pickup = PickupRequest::where('order_id', $ready->id)->lockForUpdate()->first();
                if ($ready->status !== Order::STATUS_READY_FOR_PICKUP || ! $ready->isPaymentEligible()
                    || ! $pickup || (int) $pickup->logistics_partner_id !== (int) $partner->id
                    || $pickup->status !== PickupRequest::REQUESTED
                    || Delivery::where('order_id', $order)->exists()
                    || ! $this->riderStillActive($rider, $partner)) {
                    return $this->stale();
                }

                $originCenterId = $pickup->origin_sorting_center_id
                    ?: $rider->sorting_center_id
                    ?: $partner->mainSortingCenter()->value('id')
                    ?: $partner->sortingCenters()->where('status', 'active')->value('id');

                if ($originCenterId && $rider->sorting_center_id && (int) $rider->sorting_center_id !== (int) $originCenterId) {
                    throw ValidationException::withMessages([
                        'rider_id' => 'Choose a pickup Rider assigned to the requested origin Sorting Center.',
                    ]);
                }

                Delivery::create([
                    'order_id' => $ready->id,
                    'tracking_code' => $ready->tracking_code ?: $this->trackingCode($ready),
                    'logistics_partner_id' => $partner->id,
                    'origin_sorting_center_id' => $originCenterId,
                    'rider_id' => $rider->id,
                    'status' => Delivery::PICKUP_ASSIGNED,
                    'assigned_at' => now(),
                ]);

                $pickup->update([
                    'status' => PickupRequest::ASSIGNED,
                    'origin_sorting_center_id' => $originCenterId,
                    'assigned_at' => now(),
                ]);

                return redirect()->route('logistics.deliveries.board')
                    ->with('status', 'Pickup Rider assigned.');
            });
        } catch (QueryException $exception) {
            if (Delivery::where('order_id', $order)->exists()) {
                return $this->stale();
            }
            throw $exception;
        }
    }

    /** Sorting-center scan/receipt after the pickup rider physically hands over the parcel. */
    public function receive(Request $request, int $delivery, ParcelScanService $scanner): RedirectResponse
    {
        $partner = $this->partner($request);
        $data = $request->validate(['tracking_code' => ['required', 'string', 'max:60']]);
        $parcel = Delivery::where('logistics_partner_id', $partner->id)->findOrFail($delivery);
        if (! hash_equals((string) $parcel->tracking_code, trim($data['tracking_code']))) {
            throw ValidationException::withMessages(['tracking_code' => 'Tracking code does not match the selected delivery.']);
        }
        $centerId = (int) $parcel->origin_sorting_center_id;
        abort_unless($centerId, 422, 'Origin branch must be configured first.');
        $scanner->receive((int) $partner->id, $centerId, (int) $request->user()->id, $data['tracking_code']);
        return back()->with('status', 'Pickup parcel received and recorded at origin branch.');
    }

    /** Record the destination area selected after reading the delivery address. */
    public function sort(Request $request, int $delivery): RedirectResponse
    {
        $partner = $this->partner($request);
        $data = $request->validate(['destination_area_id' => ['required', 'integer']]);
        $area = LogisticsCoverageArea::where('logistics_partner_id', $partner->id)
            ->findOrFail($data['destination_area_id']);

        return DB::transaction(function () use ($partner, $delivery, $area) {
            $owned = Delivery::with('order')->where('logistics_partner_id', $partner->id)
                ->whereKey($delivery)->lockForUpdate()->firstOrFail();

            if ($owned->status !== Delivery::AT_SORTING_CENTER || $owned->order?->status !== Order::STATUS_TO_RECEIVE) {
                return $this->stale();
            }

            // The selected area must actually cover this order's recorded destination.
            if (! $this->areaCoversOrder($area, $owned->order)) {
                throw ValidationException::withMessages([
                    'destination_area_id' => 'The selected area does not cover the recorded delivery address.',
                ]);
            }

            $destinationCenterId = $area->sorting_center_id
                ?: $partner->mainSortingCenter()->value('id');

            $owned->update([
                'destination_area_id' => $area->id,
                'destination_sorting_center_id' => $destinationCenterId,
                'status' => Delivery::SORTED,
                'sorted_at' => now(),
            ]);

            return back()->with('status', 'Parcel sorted by destination area.');
        });
    }

    /** Assign the rider responsible for final delivery after sorting, or after a failed attempt. */
    public function assignDelivery(Request $request, int $delivery): RedirectResponse
    {
        $partner = $this->partner($request);
        $data = $request->validate(['rider_id' => ['required', 'integer']]);
        $rider = $this->activeRider($partner, (int) $data['rider_id']);

        return DB::transaction(function () use ($partner, $delivery, $rider) {
            $owned = Delivery::with(['order', 'destinationArea', 'transfers'])
                ->where('logistics_partner_id', $partner->id)
                ->whereKey($delivery)->lockForUpdate()->firstOrFail();

            if (! in_array($owned->status, [Delivery::SORTED, Delivery::DELIVERY_FAILED], true)
                || $owned->order?->status !== Order::STATUS_TO_RECEIVE
                || $owned->returnCase()->exists()
                || ! $owned->destination_area_id || ! $this->riderStillActive($rider, $partner)) {
                return $this->stale();
            }

            if ((int) $rider->coverage_area_id !== (int) $owned->destination_area_id) {
                throw ValidationException::withMessages([
                    'rider_id' => 'Choose an active Rider assigned to this parcel\'s destination area.',
                ]);
            }

            if ($owned->destination_sorting_center_id
                && (int) $owned->current_sorting_center_id !== (int) $owned->destination_sorting_center_id) {
                throw ValidationException::withMessages([
                    'rider_id' => 'Receive the parcel at its destination Sorting Center before assigning the final-delivery Rider.',
                ]);
            }

            if ($owned->destination_sorting_center_id
                && (int) $rider->sorting_center_id !== (int) $owned->destination_sorting_center_id) {
                throw ValidationException::withMessages([
                    'rider_id' => 'Choose a Rider assigned to the destination Sorting Center.',
                ]);
            }

            if ($owned->transfers->contains(fn ($transfer) => $transfer->status === ParcelTransfer::IN_TRANSIT)) {
                throw ValidationException::withMessages([
                    'rider_id' => 'This parcel is still in transit between Sorting Centers.',
                ]);
            }

            $owned->update([
                'rider_id' => $rider->id,
                'delivery_rider_id' => $rider->id,
                'status' => Delivery::DELIVERY_ASSIGNED,
                'delivery_assigned_at' => now(),
                'delivery_attempts' => $owned->delivery_attempts + 1,
            ]);

            return back()->with('status', 'Delivery Rider assigned for the destination area.');
        });
    }

    /** Dispatch a sorted parcel from its current center to the branch that owns the destination area. */
    public function dispatchTransfer(Request $request, int $delivery): RedirectResponse
    {
        $partner = $this->partner($request);
        $data = $request->validate([
            'to_sorting_center_id' => ['required', 'integer'],
            'transit_rider_id' => ['nullable', 'integer'],
        ]);

        return DB::transaction(function () use ($partner, $delivery, $data) {
            $owned = Delivery::with(['order', 'transfers'])
                ->where('logistics_partner_id', $partner->id)
                ->whereKey($delivery)->lockForUpdate()->firstOrFail();

            if ($owned->status !== Delivery::SORTED
                || ! $owned->current_sorting_center_id
                || ! $owned->destination_sorting_center_id
                || (int) $owned->current_sorting_center_id === (int) $owned->destination_sorting_center_id) {
                return $this->stale();
            }

            if ($owned->transfers->contains(fn ($transfer) => $transfer->status === ParcelTransfer::IN_TRANSIT)) {
                return back()->withErrors(['delivery' => 'This parcel already has an active center transfer.']);
            }

            $from = SortingCenter::where('logistics_partner_id', $partner->id)->findOrFail($owned->current_sorting_center_id);
            $to = SortingCenter::where('logistics_partner_id', $partner->id)
                ->where('status', 'active')
                ->findOrFail((int) $data['to_sorting_center_id']);

            if ((int) $from->id === (int) $to->id) {
                throw ValidationException::withMessages([
                    'to_sorting_center_id' => 'Choose a different next Sorting Center.',
                ]);
            }

            $transitRider = null;
            if (! empty($data['transit_rider_id'])) {
                $transitRider = $this->activeRider($partner, (int) $data['transit_rider_id']);
                if ((int) $transitRider->sorting_center_id !== (int) $from->id) {
                    throw ValidationException::withMessages([
                        'transit_rider_id' => 'Transit Rider must belong to the dispatching branch.',
                    ]);
                }
            }

            ParcelTransfer::create([
                'delivery_id' => $owned->id,
                'from_sorting_center_id' => $from->id,
                'transit_rider_id' => $transitRider?->id,
                'to_sorting_center_id' => $to->id,
                'status' => ParcelTransfer::IN_TRANSIT,
                'dispatched_at' => now(),
            ]);

            return back()->with('status', 'Parcel dispatched to '.$to->name.'.');
        });
    }

    /** Destination branch scan that closes the active hub-to-hub transfer. */
    public function receiveTransfer(Request $request, int $transfer, ParcelScanService $scanner): RedirectResponse
    {
        $partner = $this->partner($request);
        $data = $request->validate(['tracking_code' => ['required', 'string', 'max:60']]);
        $parcelTransfer = ParcelTransfer::with('delivery')->findOrFail($transfer);
        abort_unless((int) $parcelTransfer->delivery?->logistics_partner_id === (int) $partner->id, 404);
        abort_unless($parcelTransfer->status === ParcelTransfer::IN_TRANSIT, 422, 'Transfer is not in transit.');
        if (! hash_equals((string) $parcelTransfer->delivery->tracking_code, trim($data['tracking_code']))) {
            throw ValidationException::withMessages(['tracking_code' => 'Tracking code does not match this transfer.']);
        }
        $scanner->receive((int) $partner->id, (int) $parcelTransfer->to_sorting_center_id,
            (int) $request->user()->id, $data['tracking_code']);
        return back()->with('status', 'Transfer received and recorded at destination branch.');
    }

    public static function readyFor(LogisticsPartner $partner): Collection
    {
        return Order::with(['seller', 'items.product', 'pickupRequest.originSortingCenter'])
            ->where('status', Order::STATUS_READY_FOR_PICKUP)
            ->paymentEligible()
            ->whereHas('pickupRequest', function ($query) use ($partner) {
                $query->where('logistics_partner_id', $partner->id)
                    ->where('status', PickupRequest::REQUESTED);
            })
            ->whereNotIn('id', Delivery::select('order_id'))
            ->latest()->get();
    }

    private function areaCoversOrder(LogisticsCoverageArea $area, Order $order): bool
    {
        $coverage = app(PhilippineDeliveryCoverage::class);
        $partner = LogisticsPartner::findOrFail($area->logistics_partner_id);
        return $coverage->coversOrder($area, $order)
            && $coverage->availableForPartner($area, $partner);
    }

    private function activeRider(LogisticsPartner $partner, int $riderId): Rider
    {
        return Rider::where('logistics_partner_id', $partner->id)
            ->where('status', 'active')->whereNotNull('email')->whereNotNull('password')
            ->findOrFail($riderId);
    }

    private function riderStillActive(Rider $rider, LogisticsPartner $partner): bool
    {
        return Rider::whereKey($rider->id)->where('logistics_partner_id', $partner->id)
            ->where('status', 'active')->whereNotNull('email')->whereNotNull('password')->exists();
    }

    private function trackingCode(Order $order): string
    {
        $date = ($order->created_at ?? now())->format('Ymd');
        return 'SHP-'.$date.'-'.str_pad((string) $order->id, 6, '0', STR_PAD_LEFT);
    }

    private function partner(Request $request): LogisticsPartner
    {
        return LogisticsPartner::with('coverageAreas')->where('user_id', $request->user()->id)->first()
            ?? abort(403, 'Logistics profile unavailable.');
    }

    private function stale(): RedirectResponse
    {
        return redirect()->route('logistics.deliveries.board')
            ->withErrors(['delivery' => 'This delivery changed or is not eligible. Refresh and review it.']);
    }
}

===== app/Services/LogisticsRoutingService.php =====

<?php

namespace App\Services;

use App\Models\Buyer\Order\Order;
use App\Models\Logistics\SortingCenter;
use App\Models\LogisticsCoverageArea;
use App\Models\LogisticsPartner;
use App\Models\Seller;
use Illuminate\Support\Collection;

class LogisticsRoutingService
{
    public function __construct(private readonly PhilippineDeliveryCoverage $coverage)
    {
    }

    public function eligiblePartners(Order $order): Collection
    {
        return LogisticsPartner::query()
            ->whereHas('user', fn ($query) => $query->where('status', 'approved'))
            ->with(['user', 'coverageAreas.sortingCenter', 'sortingCenters'])
            ->orderBy('company_name')
            ->get()
            ->filter(fn (LogisticsPartner $partner) => $this->destinationArea($partner, $order) !== null
                && $order->seller
                && $this->originCenter($partner, $order->seller) !== null)
            ->values();
    }

    /**
     * Partners that can pick up from this Seller and deliver to a destination, before any Order exists.
     * Used by Checkout so the Buyer only sees couriers that really cover the address.
     *
     * @param array<string, mixed> $destination delivery_* attributes (region/province/municipality names, address)
     */
    public function eligiblePartnersFor(Seller $seller, array $destination): Collection
    {
        // An unsaved Order only carries the destination fields the coverage rule reads.
        $draft = new Order();
        $draft->forceFill($destination);
        $draft->setRelation('seller', $seller);

        return $this->eligiblePartners($draft);
    }

    public function destinationArea(LogisticsPartner $partner, Order $order): ?LogisticsCoverageArea
    {
        return $partner->coverageAreas->first(
            fn (LogisticsCoverageArea $area) => $this->coverage->coversOrder($area, $order)
                && $this->coverage->availableForPartner($area, $partner)
        );
    }

    public function originCenter(LogisticsPartner $partner, Seller $seller): ?SortingCenter
    {
        $areas = $partner->coverageAreas()->with('sortingCenter')->get();
        $matched = $areas->first(fn (LogisticsCoverageArea $area) => $this->coverage->matches(
            $area,
            $seller->province_name,
            (string) $seller->municipality_name,
            $seller->region_name,
        ) && $this->coverage->availableForPartner($area, $partner));

        if (! $matched) {
            return null;
        }

        // A matching inactive assigned branch must never reroute to the main hub.
        // Only legacy, unassigned coverage belongs to the active main hub.
        return $matched->sorting_center_id !== null
            ? $matched->sortingCenter
            : $partner->sortingCenters()->where('status', 'active')->where('is_main', true)->first();
    }

    public function trackingCode(Order $order): string
    {
        if ($order->tracking_code) {
            return $order->tracking_code;
        }

        $date = ($order->created_at ?? now())->format('Ymd');
        return 'SHP-'.$date.'-'.str_pad((string) $order->id, 6, '0', STR_PAD_LEFT);
    }

<?php

namespace App\Services;

use App\Models\Buyer\Order\Order;
use App\Models\Logistics\Delivery;
use App\Models\Logistics\ParcelTransfer;
use App\Models\Logistics\PickupRequest;
use App\Models\Logistics\SortingCenter;
use Illuminate\Support\Facades\DB;
use Illuminate\Validation\ValidationException;

/** One authoritative hand-off rule for Web and Mobile branch scan clients. */
class ParcelScanService
{
    /**
     * Read-only eligibility check for a parcel awaiting physical handoff.
     * Receive always revalidates in its own locked transaction; preview is not authorization.
     */
    public function inspect(int $partnerId, int $centerId, string $trackingCode): array
    {
        $code = trim($trackingCode);
        if (! preg_match('/^SHP-[0-9]{8}-[0-9]{6,}$/', $code)) {
            throw ValidationException::withMessages(['tracking_code' => 'Scan a valid ShopHop shipping label.']);
        }

        $center = SortingCenter::where('logistics_partner_id', $partnerId)
            ->where('status', 'active')->findOrFail($centerId);
        $delivery = Delivery::with('order')->where('logistics_partner_id', $partnerId)
            ->where('tracking_code', $code)->first();
        if (! $delivery) {
            throw ValidationException::withMessages(['tracking_code' => 'Parcel not found in your Logistics network.']);
        }

        if (! $delivery->order || $delivery->order->status !== Order::STATUS_TO_RECEIVE) {
            throw ValidationException::withMessages(['tracking_code' => 'Parcel is not eligible for branch receipt.']);
        }

        if ($delivery->status === Delivery::PICKED_UP
            && (int) $delivery->origin_sorting_center_id === $centerId) {
            $action = 'Receive pickup from Rider';
            $event = 'pickup_received';
        } elseif ($delivery->status === Delivery::SORTED) {
            $transfer = ParcelTransfer::where('delivery_id', $delivery->id)
                ->where('status', ParcelTransfer::IN_TRANSIT)->first();
            if (! $transfer || (int) $transfer->to_sorting_center_id !== $centerId
                || ($transfer->transit_rider_id && ! $transfer->departed_at)) {
                throw ValidationException::withMessages(['tracking_code' => 'No departed branch transfer is awaiting receipt here.']);
            }
            $action = 'Receive inter-branch transfer';
            $event = 'transfer_received';
        } else {
            throw ValidationException::withMessages(['tracking_code' => 'Wrong branch, duplicate scan, or parcel not ready for receipt.']);
        }

        // Intentionally exclude recipient PII from scanner preview.
        return [
            'tracking_code' => $delivery->tracking_code,
            'branch' => $center->name,
            'delivery_status' => $delivery->status,
            'event' => $event,
            'next_action' => $action,
            'can_receive' => true,
        ];
    }

    public function receive(int $partnerId, int $centerId, int $operatorId, string $trackingCode, ?int $staffId = null): array
    {
        $code = trim($trackingCode);
        if (! preg_match('/^SHP-[0-9]{8}-[0-9]{6,}$/', $code)) {
            throw ValidationException::withMessages(['tracking_code' => 'Scan a valid ShopHop parcel label, not an external URL.']);
        }

        return DB::transaction(function () use ($partnerId, $centerId, $operatorId, $code, $staffId) {
            $center = SortingCenter::where('logistics_partner_id', $partnerId)
                ->where('status', 'active')->findOrFail($centerId);
            $delivery = Delivery::with('order')->where('logistics_partner_id', $partnerId)
                ->where('tracking_code', $code)->lockForUpdate()->firstOrFail();
            if (! $delivery->order || $delivery->order->status !== Order::STATUS_TO_RECEIVE) {
                throw ValidationException::withMessages(['tracking_code' => 'Parcel is not currently eligible for branch receipt.']);
            }

            if ($delivery->status === Delivery::PICKED_UP) {
                if (! $delivery->origin_sorting_center_id || (int) $delivery->origin_sorting_center_id !== $centerId) {
                    throw ValidationException::withMessages(['sorting_center_id' => 'This parcel belongs to a different pickup branch.']);
                }
                $delivery->update([
                    'status' => Delivery::AT_SORTING_CENTER,
                    'current_sorting_center_id' => $centerId,
                    'at_sorting_center_at' => now(),
                ]);
                PickupRequest::where('order_id', $delivery->order_id)->update([
                    'status' => PickupRequest::RECEIVED,
                    'received_at' => now(),
                ]);
                $type = 'pickup_received';
                $transferId = null;
            } elseif ($delivery->status === Delivery::SORTED) {
                $transfer = ParcelTransfer::where('delivery_id', $delivery->id)
                    ->where('status', ParcelTransfer::IN_TRANSIT)->lockForUpdate()->first();
                if (! $transfer || (int) $transfer->to_sorting_center_id !== $centerId
                    || ($transfer->transit_rider_id && ! $transfer->departed_at)) {
                    throw ValidationException::withMessages([
                        'sorting_center_id' => 'No departed transfer is awaiting receipt at this branch.',
                    ]);
                }
                $transfer->update(['status' => ParcelTransfer::RECEIVED, 'received_at' => now()]);
                $delivery->update(['current_sorting_center_id' => $centerId]);
                $type = 'transfer_received';
                $transferId = $transfer->id;
            } else {
                throw ValidationException::withMessages([
                    'tracking_code' => 'Parcel already received or currently at a different workflow stage.',
                ]);
            }

            DB::table('parcel_scan_events')->insert([
                'delivery_id' => $delivery->id,
                'sorting_center_id' => $center->id,
                'operator_user_id' => $operatorId,
                'operator_staff_id' => $staffId,
                'parcel_transfer_id' => $transferId,
                'event_type' => $type,
                'scanned_at' => now(),
                'created_at' => now(),
                'updated_at' => now(),
            ]);

            return [
                'delivery_id' => $delivery->id,
                'tracking_code' => $delivery->tracking_code,
                'event' => $type,
                'delivery_status' => $delivery->status,
                'branch' => $center->name,
            ];
        }, 3);
    }
}

<script>
document.addEventListener('DOMContentLoaded', function () {
    const shell = document.getElementById('branchShell');
    const sidebar = document.getElementById('branchSidebar');
    const toggle = document.getElementById('branchMobileToggle');
    const overlay = document.getElementById('branchMobileOverlay');
    const sidebarClose = document.getElementById('branchSidebarClose');

    function openSidebar() {
        shell?.setAttribute('data-mobile-open', 'true');
        if (overlay) overlay.hidden = false;
        document.body.style.overflow = 'hidden';
    }
    function closeSidebar() {
        shell?.setAttribute('data-mobile-open', 'false');
        if (overlay) overlay.hidden = true;
        document.body.style.overflow = '';
    }
    toggle?.addEventListener('click', openSidebar);
    overlay?.addEventListener('click', closeSidebar);
    sidebarClose?.addEventListener('click', closeSidebar);
    document.querySelectorAll('#branchSidebar a, #branchSidebar button[type="submit"]').forEach(function (el) {
        el.addEventListener('click', function () { if (window.innerWidth < 1024) closeSidebar(); });
    });

    /* Desktop collapse */
    const key = 'shophop_branch_sidebar_collapsed';
    function setCollapsed(collapsed) {
        const apply = window.innerWidth >= 1024 && collapsed;
        sidebar?.setAttribute('data-collapsed', apply ? 'true' : 'false');
        localStorage.setItem(key, apply ? '1' : '0');
    }
    setCollapsed(localStorage.getItem(key) === '1');
    document.getElementById('branchDesktopCollapse')?.addEventListener('click', function () { setCollapsed(true); });
    document.getElementById('branchDesktopExpand')?.addEventListener('click', function (e) { e.preventDefault(); setCollapsed(false); });
    window.addEventListener('resize', function () {
        if (window.innerWidth >= 1024) { closeSidebar(); setCollapsed(localStorage.getItem(key) === '1'); }
        else { setCollapsed(false); }
    });

    /* Command palette */
    const searchTrigger = document.getElementById('branchSearchTrigger');
    const searchPalette = document.getElementById('branchSearchPalette');
    const searchBackdrop = document.getElementById('branchSearchBackdrop');
    const commandInput = document.getElementById('branchCommandInput');
    const commandItems = Array.from(document.querySelectorAll('[data-branch-command-item]'));
    const commandEmpty = document.getElementById('branchCommandEmpty');

    function filterCommands() {
        const q = (commandInput?.value || '').trim().toLowerCase();
        let visible = 0;
        commandItems.forEach(function (item) {
            const show = q === '' || (item.getAttribute('data-search-text') || '').includes(q);
            item.hidden = !show;
            if (show) visible++;
        });
        if (commandEmpty) commandEmpty.hidden = visible > 0;
    }
    function openPalette() {
        if (!searchPalette) return;
        searchPalette.hidden = false;
        searchPalette.setAttribute('aria-hidden', 'false');
        document.body.style.overflow = 'hidden';
        setTimeout(function () { commandInput?.focus(); commandInput?.select(); }, 0);
    }
    function closePalette() {
        if (!searchPalette) return;
        searchPalette.hidden = true;
        searchPalette.setAttribute('aria-hidden', 'true');
        document.body.style.overflow = '';
        if (commandInput) commandInput.value = '';
        filterCommands();
    }
    searchTrigger?.addEventListener('click', openPalette);
    searchBackdrop?.addEventListener('click', closePalette);
    commandInput?.addEventListener('input', filterCommands);
    commandInput?.addEventListener('keydown', function (e) {
        if (e.key !== 'Enter') return;
        const first = commandItems.find(function (i) { return !i.hidden; });
        if (first) window.location.href = first.href;
    });
    document.addEventListener('keydown', function (e) {
        if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'k') { e.preventDefault(); openPalette(); return; }
        if (e.key === 'Escape') {
            if (searchPalette && !searchPalette.hidden) { closePalette(); return; }
            if (shell?.getAttribute('data-mobile-open') === 'true') closeSidebar();
        }
    });
    filterCommands();

    /* Confirm modal */
    const confirmModal = document.getElementById('branchConfirmModal');
    const confirmTitle = document.getElementById('branchConfirmTitle');
    const confirmBody = document.getElementById('branchConfirmBody');
    const confirmAccept = document.getElementById('branchConfirmAccept');
    let confirmHandler = null;
    let previousFocus = null;

    function openConfirm(o) {
        if (!confirmModal) return;
        confirmTitle.textContent = o.title || 'Confirm action';
        confirmBody.textContent = o.message || 'Please confirm to continue.';
        confirmAccept.textContent = o.confirmLabel || 'Continue';
        confirmAccept.className = (o.danger ? 'sh-btn-danger' : 'sh-btn') + ' px-5';
        confirmHandler = o.onConfirm || null;
        previousFocus = document.activeElement;
        confirmModal.hidden = false;
        document.body.style.overflow = 'hidden';
    }
    function closeConfirm() {
        if (!confirmModal) return;
        confirmModal.hidden = true;
        document.body.style.overflow = (searchPalette && !searchPalette.hidden) ? 'hidden' : '';
        confirmHandler = null;
        previousFocus?.focus?.();
        previousFocus = null;
    }
    document.querySelectorAll('[data-confirm-close]').forEach(function (b) { b.addEventListener('click', closeConfirm); });
    confirmAccept?.addEventListener('click', function () {
        if (typeof confirmHandler === 'function') { const run = confirmHandler; closeConfirm(); run(); }
    });
    document.addEventListener('submit', function (event) {
        const form = event.target.closest('form[data-confirm-title]');
        if (!form || form.dataset.confirmed === '1') { if (form) form.dataset.confirmed = '0'; return; }
        event.preventDefault();
        openConfirm({
            title: form.dataset.confirmTitle,
            message: form.dataset.confirmMessage || 'Please confirm to continue.',
            confirmLabel: form.dataset.confirmButton || 'Continue',
            danger: form.dataset.confirmDanger === '1',
            onConfirm: function () { form.dataset.confirmed = '1'; form.requestSubmit(); }
        });
    }, true);
    document.addEventListener('keydown', function (e) {
        if (!confirmModal || confirmModal.hidden) return;
        if (e.key === 'Escape') { e.preventDefault(); closeConfirm(); }
    }, true);
});
</script>

(function () {

    const checkAll = document.querySelector('[data-check-all]');
    const selected = form.querySelector('[data-selected]');

    function sync() {
        const n = boxes.filter(function (b) { return b.checked; }).length;
        selected.textContent = n;
        form.hidden = n === 0;
        if (checkAll) {
            checkAll.checked = boxes.length > 0 && n === boxes.length;
            checkAll.indeterminate = n > 0 && n < boxes.length;
        }
    }

    boxes.forEach(function (b) { b.addEventListener('change', sync); });
    checkAll && checkAll.addEventListener('change', function () {
        boxes.forEach(function (b) { b.checked = checkAll.checked; });
        sync();
    });

    sync();
})();
